CREATE OR REPLACE PACKAGE BODY OFSA_SB.MS_EX_OFSA_VADE_BOZUM_ZARARI IS

PROCEDURE MS_EX_OFSA_VADE_BOZUM_ZARARI (SECIM NUMBER)IS

    ERROR_LOCATION     NUMBER;
    PID                NUMBER;
    PDATE              DATE;
    AYBASI             DATE;
    PASOFDATE          VARCHAR2(8);
    PCRITICAL_LVL      NUMBER := 1; --1:BLOCKER,2:CRITICAL,3:TRIVIAL,9:NO PROBLEM
    SEGM_TARIH         DATE;
BEGIN

    ERROR_LOCATION := 1;

    SELECT PRM_VALUE, TO_DATE(PRM_VALUE, PRM_FORMAT),TO_DATE(MONTH_BEGIN, PRM_FORMAT),TO_DATE(LAST_SEGM_DATE, PRM_FORMAT)
      INTO PASOFDATE, PDATE,AYBASI,SEGM_TARIH
      FROM OFSA_SB.SY_SYSTEM_PARAMETERS T
      WHERE PARAMETER = 'OFSA_SB_DATE';

      IF PDATE<=SEGM_TARIH THEN --SEGMENTASYON ATAMASI
      SEGM_TARIH:=SEGM_TARIH;
      ELSE
      SEGM_TARIH:=PDATE;
      END IF;

    PID := OFSA_SB.MN_UTILITY.INSERT_PLOG(PMODULE      => 'MS_EX_OFSA_SB_VADE_BOZUM_ZARARI',
                                        PSUBMODULE   => 'EX_OFSA_SB_VADE_BOZUM_ZARARI',
                                        PROJECT_NAME => 'OFSA_SB',
                                        PASOFDATE    => PDATE,
                                        PCRITICAL    => PCRITICAL_LVL);
   /*****************************************************************************************************************************/
    

    --EXECUTE IMMEDIATE 'ALTER SESSION ENABLE PARALLEL DML';

    IF SECIM=0 THEN --Eger 0 seçilirse günlük,0 disinda bir veri seçilirse aylik yükler
    AYBASI:=PDATE;
    END IF; 

    EXECUTE IMMEDIATE 'TRUNCATE TABLE OFSA_SB.VADE_BOZUM_TEMP';
    
    
    INSERT INTO VADE_BOZUM_TEMP 
    (
    SELECT U.CUSTOMER_CODE,
           U.CUSTOMER_ORG_CODE,
           U.STATUS,
           U.STATE,
           U.RECORD_TYPE,
           U.PRODUCT_PROPERTIES,
           U.USAGE_TYPE,
           U.CLOSE_DATE,
           U.IS_SENT_HTI,
           U.CREDIT_NO
    FROM CCS.CRD_CREDIT_USAGE@CBRAS_PRODUX U
    WHERE U.STATUS = '1'
    AND U.STATE IN ('2', '5')
    AND U.RECORD_TYPE = 'U'
    AND U.PRODUCT_PROPERTIES NOT IN ('GUN_ICI', 'TAHSILAT') -- BU KREDILER HARIÇ
    AND U.USAGE_TYPE IN ('2', '4') -- ISKONTO HARIÇ (SADECE SPOT(2) VE TTK(4)
    AND NVL(U.PRODUCT_PROPERTIES, 'XX') != 'EXIM' -- EXIM HARIÇ
    AND TO_DATE(U.CLOSE_DATE, 'yyyymmdd') BETWEEN AYBASI AND PDATE
    AND NVL(U.IS_SENT_HTI,0)=0 -- TAKIBE GIDENLER HARIÇ
    AND U.CREDIT_NO NOT IN --MUAFIYET MUSTERILERI CIKARILIYOR A.Y.
    (
    SELECT DISTINCT CREDIT_NO
    FROM BLACKLIST
    WHERE KAYNAK = 'VADE_BOZUM_KAR_ZARAR'
    
    )
    );
    
    COMMIT;
  
    ---------------------------------------------------------------------------------------------------------------
    EXECUTE IMMEDIATE 'TRUNCATE TABLE OFSA_SB.OFSA_EX_VADE_BOZUM_ZARARI';
     
    INSERT
      INTO OFSA_SB.OFSA_EX_VADE_BOZUM_ZARARI

     (
    SELECT PDATE AS AS_OF_DATE,
      U.CUSTOMER_CODE AS MUS_NO,
      OFSA_SB.MN_UTILITY.MAP_OFSA_ISKOLU (BUSINESS_SEGMENT_CODE   => SEG.BUSINESS_SEGMENT_CODE) AS ISKOLU,
      SEG.BUSINESS_SEGMENT_CODE AS SEGMENT,
      CASE WHEN TO_NUMBER(U.CUSTOMER_ORG_CODE) IN ('6', '220', '372', '122') THEN -1
                ELSE TO_NUMBER(O.PARENT_CODE) END AS BOLGE,
      TO_NUMBER(U.CUSTOMER_ORG_CODE) AS SUBE,
      NVL(A.REPRESENTATIVE_FILE_NO, 999999) AS MT,
      A.PRODUCT_TYPE AS URUN_KODU,
      A.PRODUCT_NAME AS URUN,
      CASE WHEN A.IS_INDEXED = '1' THEN A.INDEX_CURRENCY ELSE A.CURRENCY_CODE END AS PB,
      'VADE_BOZUM_KAR_ZARAR' AS KAYNAK,
      ROUND(SUM((CASE WHEN A.IS_INDEXED = '1' THEN R.EVALUATION_SELL_RATE ELSE R.EVALUATION_BUY_RATE END) * A.MAIN_BALANCE_ORG * 1 / 100),0) AS TUTAR
    FROM VADE_BOZUM_TEMP  U,
      SUMMARY_PROD.GL_ASSET               A,
      OLAP.FC_EVALUATION_RATES            R,
      OLAP.DM_ORGANIZATION                O,
      OLAP.FC_CUSTOMER                    C,
      OLAP.FC_CUSTOMER                    SEG
    WHERE A.AS_OF_DATE BETWEEN last_day(add_months(PDATE,-1))+1 AND PDATE
    AND A.AS_OF_DATE=TO_DATE(U.CLOSE_DATE, 'yyyymmdd')-1
    AND C.AS_OF_DATE = PDATE
    AND SEG.AS_OF_DATE = SEGM_TARIH
    AND R.AS_OF_DATE = TO_DATE(U.CLOSE_DATE, 'yyyymmdd')-1
    AND A.ACCOUNT_NO = U.CREDIT_NO
    AND TO_NUMBER(A.CUSTOMER_CODE)=TO_NUMBER(U.CUSTOMER_CODE)
    AND A.AS_OF_DATE = TO_DATE(U.CLOSE_DATE, 'yyyymmdd')-1
    AND A.MATURITY_DATE - TO_DATE(U.CLOSE_DATE, 'yyyymmdd') >= 61 -- VADE ILE KAPAMA TARIHI 60 GÜNDEN BÜYÜKLER
    AND TO_NUMBER(C.CUSTOMER_CODE)=TO_NUMBER(U.CUSTOMER_CODE)
    AND TO_NUMBER(SEG.CUSTOMER_CODE)=TO_NUMBER(U.CUSTOMER_CODE)
    AND TO_NUMBER(O.GL_ORG_CODE) = TO_NUMBER(A.BRANCH_CODE)
    AND A.CUSTOMER_CODE=C.CUSTOMER_CODE
    AND A.CUSTOMER_CODE=seg.CUSTOMER_CODE
    AND A.MAIN_BALANCE_ORG<>0
    AND R.CURRENCY_CODE =(CASE WHEN A.IS_INDEXED = '1' THEN A.INDEX_CURRENCY ELSE A.CURRENCY_CODE END)
    AND NVL(A.INTEREST_TYPE, 'XX') != 'D' -- DEGISKEN FAIZLILER HARIÇ
    AND NVL(A.AGING_LATE_DAY,0) < 31 -- GECIKME GÜN SAYISI 30 DAN BÜYÜKLER HARIÇ
    GROUP BY PDATE,
        U.CUSTOMER_CODE,
      OFSA_SB.MN_UTILITY.MAP_OFSA_ISKOLU (BUSINESS_SEGMENT_CODE   => SEG.BUSINESS_SEGMENT_CODE),
        SEG.BUSINESS_SEGMENT_CODE,
        CASE WHEN TO_NUMBER(U.CUSTOMER_ORG_CODE) IN ('6', '220', '372', '122') THEN -1
                ELSE TO_NUMBER(O.PARENT_CODE) END,
        TO_NUMBER(U.CUSTOMER_ORG_CODE),
        A.REPRESENTATIVE_FILE_NO,
        A.PRODUCT_TYPE,
        A.PRODUCT_NAME,
        CASE WHEN A.IS_INDEXED = '1' THEN A.INDEX_CURRENCY ELSE A.CURRENCY_CODE END

     );

      COMMIT;

    /*******************************************************************************************************************/

   OFSA_SB.MN_UTILITY.UPDATE_PLOG(PID => PID, SUCCESS => 1, PCRITICAL => 9);

  EXCEPTION
    WHEN OTHERS THEN
      OFSA_SB.MN_UTILITY.UPDATE_PLOG(PID => PID,
                                   SUCCESS => 0,
                                   FAIL => 'POSITION: ' || ERROR_LOCATION || ' ERROR: ' || ABS(SQLCODE) || ':' || SQLERRM || ' ' || DBMS_UTILITY.FORMAT_ERROR_BACKTRACE,
                                   PCRITICAL => PCRITICAL_LVL);
         IF ABS(SQLCODE) = '08103'
         OR ABS(SQLCODE) = '12801'     
                               
      THEN
        ROLLBACK;
--        DBMS_LOCK.SLEEP(180);
          OFSA_SB.MS_EX_OFSA_VADE_BOZUM_ZARARI.MS_EX_OFSA_VADE_BOZUM_ZARARI(0);
      ELSE
      ROLLBACK;
        RAISE;
        END IF;
  END;

END;
