PROCEDURE MS_EX_OFSA_SB_MIZAN_GELIRGIDER (SECIM NUMBER) IS

    ERROR_LOCATION     NUMBER;

    AYBASI             DATE;
    PASOFDATE          VARCHAR2(8);
    PCRITICAL_LVL      NUMBER := 1; --1:BLOCKER,2:CRITICAL,3:TRIVIAL,9:NO PROBLEM
    SEGM_TARIH         DATE;
    READY              NUMBER;
BEGIN

    ERROR_LOCATION := 1;

    SELECT PRM_VALUE, TO_DATE(PRM_VALUE, 'YYYYMMDD'),TO_DATE(MONTH_BEGIN, 'YYYYMMDD'),TO_DATE(LAST_SEGM_DATE, 'YYYYMMDD')
      INTO PASOFDATE, PDATE,AYBASI,SEGM_TARIH
      FROM OFSA_SB.SY_SYSTEM_PARAMETERS T
      WHERE PARAMETER = 'OFSA_SB_DATE';

      SEGM_TARIH:=SEGM_TARIH;
      ELSE
      SEGM_TARIH:=PDATE;
      END IF;

        SELECT count(*)
        into READY
        FROM OFSA_DAILY.OFSA_D_CLEARING_REPORT@QADATA_LINK
        where as_of_date=PDATE;
        exit when READY>4000;
        sleep(300000);--5dk
        end loop;

        SELECT count(*)
        into READY
        FROM BI_APP.VW_OFSA_FACT_LOAN_CONTRACT@QADATA_LINK

        sleep(300000);--5dk
        end loop;

                            ELSE (TARIM_KOMISYON_KONTROL - 0.005)
