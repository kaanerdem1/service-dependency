      ELSE
      SEGM_TARIH:=PDATE;

                      THEN LEAST( KOK5, DIS_TIC_MIS )

              FROM  
                (
                SELECT A.AS_OF_DATE,
                       A.SUBE_KODU,
                       sum(case when A.KOK < 0 then 0 ELSE A.KOK END) AS KOK,
                       sum(case when A.KOK25 < 0 then 0 ELSE A.KOK25 END) AS KOK25,
                       sum(case when A.KOK5 < 0 then 0 ELSE A.KOK5 END) AS KOK5,
                       sum(B.DIGER_MIS) AS DIGER_MIS,
                       sum(case when KOK < 0 THEN 0 ELSE B.DIS_TIC_MIS END) AS DIS_TIC_MIS,

                     A.AS_OF_DATE=PDATE AND 
                     nvl(A.PRIVATE_BRANCH_FLAG,0)=B.PRIVATE_BRANCH_FLAG AND
                     A.PRIVATE_BRANCH_FLAG=1
                     AND NVL(A.GOLGE_FLAG,0) = NVL(B.GOLGE_FLAG,0)
                    )

            WHEN MATCHED THEN
              UPDATE SET A.AYLIK_FAALIYET_KARI_LINEER=A.FAALIYET_KARI_LINEER-NVL(B.FAALIYET_KARI_LINEER,0),
                         A.AYLIK_KRSLK_ONCS_NET_KAR_LNR=A.KARSILIK_ONCESI_NET_KAR_LNR-NVL(B.KARSILIK_ONCESI_NET_KAR_LNR,0),
                         A.AYLIK_KRSLK_ONCS_NET_KAR_IRR=A.KARSILIK_ONCESI_NET_KAR_IRR-NVL(B.KARSILIK_ONCESI_NET_KAR_IRR,0),
                         A.AYLIK_FAALIYET_KARI_IRR=A.FAALIYET_KARI_IRR-NVL(B.FAALIYET_KARI_IRR,0),
                         A.AYLIK_PERFORMANS_KARI_LINEER=A.PERFORMANS_KARI_LINEER-NVL(B.PERFORMANS_KARI_LINEER,0),
                         A.AYLIK_PERFORMANS_KARI_IRR=A.PERFORMANS_KARI_IRR-NVL(B.PERFORMANS_KARI_IRR,0);
      COMMIT;  
    /*******************************************************************************************************************/

   OFSA_SB.MN_UTILITY.UPDATE_PLOG(PID => PID, SUCCESS => 1, PCRITICAL => 9);

  EXCEPTION
    WHEN OTHERS THEN
      OFSA_SB.MN_UTILITY.UPDATE_PLOG(PID => PID,
                                   SUCCESS => 0,
                                   FAIL => 'POSITION: ' || ERROR_LOCATION || ' ERROR: ' || ABS(SQLCODE) || ':' || SQLERRM || ' ' || DBMS_UTILITY.FORMAT_ERROR_BACKTRACE,
                                   PCRITICAL => PCRITICAL_LVL);
      ROLLBACK;
      RAISE;
  END;

END;
