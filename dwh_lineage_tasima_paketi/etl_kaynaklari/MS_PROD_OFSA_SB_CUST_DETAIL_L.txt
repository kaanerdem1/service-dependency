      /* ------------==============PASİF TEKLİF ORANI===================
        -     Riski bulunan müşteriler içinden risk kodu ?01? veya ?02? olup ?Yapılandırma ve Analiz? kapsamında olmayanlar,

      INSERT
        INTO OFSA_SB.TEMP_PASIF_TEKLIF_ORANI
      (
          SELECT * FROM
          (SELECT
          RISK.TARIH,
          CASE
            WHEN TT.STATE_NAME = 'Aktif' THEN 0
              WHEN CASE WHEN ROW_NUMBER () OVER(PARTITION BY ORGANIZE.SUBE_KODU ORDER BY NVL(RISK.TOPLAM_RISK,0) DESC ) <= 25 THEN 1 ELSE 0 END = 1 THEN 1
                ELSE 1 END AS REVIZE_FLG,
          CASE
            WHEN MÜŞTERI.KAR_MERKEZI IN ('2-Ticari/Kurumsal','3-KOBİ') AND (RISK.TOPLAM_RISK >= 100000 )  THEN 1
              WHEN MÜŞTERI.KAR_MERKEZI NOT IN ('2-Ticari/Kurumsal','3-KOBİ') AND (RISK.TOPLAM_RISK >= 50000 ) THEN 1
                ELSE 0 END AS RISK_KRITERI_FLG,
          CASE WHEN ROW_NUMBER () OVER(PARTITION BY ORGANIZE.SUBE_KODU ORDER BY NVL(RISK.TOPLAM_RISK,0) DESC ) <= 25 THEN 1 ELSE 0 END AS ILK_25_FLG,
          ORGANIZE.BOLGE_ADI,
          ORGANIZE.SUBE_KODU,
          ORGANIZE.SUBE_ADI,
          RISK.MUSTERI_NO,
          RISK.MUSTERI_ADI,
          TT.ALLOTMENT_NO AS SON_TAHSIS_NO,
          CASE WHEN (NVL(TT.ACR_REMAKE,'NO') = 'YES' OR NVL(TT.LIQUIDATION,'0') = '1') THEN 'Evet' ELSE 'Hayır' END AS TASFIYE,
          TT.MODÜL,
          TT.TEKLIF_TÜRÜ,
          TT.STATE_NAME AS DURUM,
          TT.MATURITY_DATE AS VADE_TARIHI,
          NVL(TT.APPROVE_CHAIR,'XX') APPROVE_CHAIR,
          CASE
          WHEN TT.APPROVE_CHAIR LIKE 'BK%' THEN 'Bölge Kredi Komitesi'
          WHEN TT.APPROVE_CHAIR LIKE 'BT%' THEN 'Bölge Krediler Müdürü'
          WHEN TT.APPROVE_CHAIR LIKE 'BL%' THEN 'Bölge Müdürü'
          WHEN TT.APPROVE_CHAIR LIKE 'BOTM%' THEN 'Bölge Krediler Müdürü'
          WHEN TT.APPROVE_CHAIR LIKE 'SBM%' THEN 'Şube Müdürü'
          WHEN TT.APPROVE_CHAIR IN ('GM','GMD','GM1') THEN 'Genel Müdür'
          WHEN TT.APPROVE_CHAIR IN ('GMY', 'GMY1') THEN 'Tahsis GMY'
          WHEN TT.APPROVE_CHAIR = 'GMYKK' THEN 'GMY Kredi Komitesi'
          WHEN TT.APPROVE_CHAIR IN ('TK', 'TK1') THEN 'GM Kredi Tahsis Kurulu'
          WHEN TT.APPROVE_CHAIR IN ('MKSM', 'IKSM' ) THEN 'İşletme Krediler Müdürü'
          WHEN TT.APPROVE_CHAIR IN ('TARKR', 'TKSM' ) THEN 'Tarım Krediler Müdürü'
          WHEN TT.APPROVE_CHAIR IN ('KOBI', 'KKSM' ) THEN 'Kobi Krediler Müdürü'
          WHEN TT.APPROVE_CHAIR IN ('TKRD', 'TKRDY' ) THEN 'Ticari ve Kurumsal Krediler Müdürü'
          WHEN TT.APPROVE_CHAIR IN ('KK','KK1','KKM') THEN 'Kredi Komitesi'
          WHEN TT.APPROVE_CHAIR IN ('PKRD','PRKRD') THEN 'İşletme ve Tarım Kredileri Müdürü'
          WHEN TT.APPROVE_CHAIR IN ('YKR','YKR1') THEN 'Yönetim Kurulu'
          WHEN TT.YETKI_TÜRÜ = 'Şube Yetkili' THEN 'Şube Müdürü'
          WHEN (TT.APPROVE_CHAIR IS NULL AND TT.TEKLIF_TÜRÜ = 'Mikrofinansman Teklif') THEN 'Mikro Finansman'

          A.AS_OF_DATE AS TARIH,
          A.CUSTOMER_CODE AS MUSTERI_NO,
          A.CUSTOMER_NAME AS MUSTERI_ADI,
          SUM (CASE    WHEN (A.MAIN_GL_NO LIKE '1%' AND A.MAIN_GL_NO NOT LIKE '17%'  AND A.CURRENCY_CODE = 'TL' AND A.IS_INDEXED = 0)   THEN ROUND(A.MAIN_BALANCE*-1)
                              WHEN (A.MAIN_GL_NO LIKE '1%' AND A.MAIN_GL_NO NOT LIKE '17%' AND A.CURRENCY_CODE <> 'TL')                                 THEN ROUND(A.MAIN_BALANCE*-1)
                              WHEN A.MAIN_GL_NO LIKE '1%' AND A.MAIN_GL_NO NOT LIKE '17%'  THEN ROUND(A.MAIN_BALANCE_ORG*-1*INDEX_EVALUATION_RATE)
                              ELSE 0
                              END) AS NAKIT_RISK,
          SUM (CASE    WHEN (A.MAIN_GL_NO LIKE '9%' AND A.CURRENCY_CODE = 'TL' AND A.IS_INDEXED = 0)   THEN ROUND(A.MAIN_BALANCE*-1)
                              WHEN (A.MAIN_GL_NO LIKE '9%' AND A.CURRENCY_CODE <> 'TL')                                 THEN ROUND(A.MAIN_BALANCE*-1)
                              WHEN A.MAIN_GL_NO LIKE '9%'  THEN ROUND(A.MAIN_BALANCE_ORG*-1*INDEX_EVALUATION_RATE)
                              ELSE 0
                              END) AS GN_RISK,
          SUM (CASE    WHEN (A.MAIN_GL_NO NOT LIKE '17%'  AND A.CURRENCY_CODE = 'TL' AND A.IS_INDEXED = 0)   THEN ROUND(A.MAIN_BALANCE*-1)
                              WHEN (A.MAIN_GL_NO NOT LIKE '17%' AND A.CURRENCY_CODE <> 'TL')                                 THEN ROUND(A.MAIN_BALANCE*-1)
                              WHEN A.MAIN_GL_NO NOT LIKE '17%'  THEN ROUND(A.MAIN_BALANCE_ORG*-1*INDEX_EVALUATION_RATE)
                              ELSE 0
                              END) AS TOPLAM_RISK
          FROM SUMMARY_PROD.GL_ASSET A
          WHERE 1=1
          AND A.CUSTOMER_CODE IN (SELECT T.CUSTOMER_CODE
                                  FROM OLAP.VW_FC_CS_RISK_CODE T
                                  WHERE T.AS_OF_DATE = PDATE_PASIFTEKLIF
                                  AND NVL(T.CUSTOMER_RISK_CODE,'01') IN ('01','02')
                                  AND NVL(T.PROCESS_TYPE,'xx') NOT IN ('Yapılandırma Ve Analiz')
                                  )
          AND A.AS_OF_DATE = PDATE_PASIFTEKLIF
          AND A.PROFIT_CENTER_CODE NOT IN ('1')
          AND SUBSTR(A.MAIN_GL_NO,1,1) IN ('1','9')
          AND NVL(A.MODULE,'xx') NOT IN ('CEK')
          AND (A.CUSTOMER_NAME NOT LIKE '%ŞEKERBANK%' OR A.CUSTOMER_NAME LIKE '%KIBRIS%')
          GROUP BY A.CUSTOMER_CODE,
                   A.AS_OF_DATE,
                   A.CUSTOMER_NAME  )  RISK

          LEFT JOIN OFSA_SB.TEMP_PASIF_TEKLIF_ISTISNA IST ON RISK.MUSTERI_NO=IST.MUS_NO AND IST.ISTISNA_BITIS_TARIHI>=PDATE_PASIFTEKLIF --Pasif teklifler için istisna tanımlaması yapılıyor. 19/07/2023 U.U.- D.K - M.K.Ç.
          LEFT JOIN ( SELECT
                        E.CUSTOMER_CODE,
                        CASE WHEN NVL(E.PRIVATE_BRANCH_FLAG,0)<>1 THEN E.MAIN_BRANCH_CODE
                                     ELSE E.OLD_BRANCH_CODE END AS MUSTERI_SUBESI,
                        SEG.PROFIT_CENTER_CODE || '-' || DECODE (SEG.PROFIT_CENTER_CODE,'1','Bireysel','2','Ticari/Kurumsal','3','KOBİ','4','Tarım','5','İşletme') AS KAR_MERKEZI,

                        E.RATE_GROUP_DATE AS RISK_KODU_TARIHI,
                        E.CCS_ARL_CREDIT_STATUS_DESC AS KREDI_STATÜSÜ,
                        E.RATE_GROUP AS RISK_KODU
                        FROM OLAP.FC_CUSTOMER E,
                             OLAP.FC_CUSTOMER SEG
                        WHERE E.AS_OF_DATE = PDATE_PASIFTEKLIF
                         AND SEG.AS_OF_DATE=SEGM_TARIH
                         AND E.CUSTOMER_CODE=SEG.CUSTOMER_CODE ) MÜŞTERI ON RISK.MUSTERI_NO = MÜŞTERI.CUSTOMER_CODE
          LEFT JOIN (SELECT DISTINCT
                            E.MAIN_BRANCH_CODE AS SUBE_KODU,
                            ORG.ORG_NAME AS SUBE_ADI,
                            CASE WHEN ORG.ORG_GROUP4 IS NOT NULL THEN 'KURUMSAL'
                                 WHEN E.PRIVATE_BRANCH_FLAG=1 AND ORG.ORG_NAME LIKE '%ÖZEL BANKACILIK%' THEN 'ÖZEL BANKACILIK'
                                  ELSE ORG.PARENT_NAME END AS BOLGE_ADI
                            FROM OLAP.FC_CUSTOMER E, OLAP.DM_ORGANIZATION ORG
                            WHERE 1=1
                           AND E.STATUS = 1 AND ORG.STATUS = 1
                            AND E.AS_OF_DATE = PDATE_PASIFTEKLIF
                            AND E.MAIN_BRANCH_CODE = ORG.ORG_CODE ) ORGANIZE ON MÜŞTERI.MUSTERI_SUBESI = ORGANIZE.SUBE_KODU
          INNER JOIN (SELECT
                        T.DT_APPROVE_DATE,
                        T.MATURITY_DATE,
                        CASE WHEN ADD_MONTHS(T.MATURITY_DATE,3) >= PDATE_PASIFTEKLIF THEN 'Aktif' ELSE 'Pasif' END AS STATE_NAME,
                        T.OID,
                        T.CUSTOMER_CODE,
                        T.ALLOTMENT_NO,
                        DECODE(T.MODULE,'ALLOTMENT','Tahsis','PROPOSAL','Teklif','DT_MAIN','Karar_Aracı',T.MODULE) AS MODÜL,
                        T.APPROVE_CHAIR,
                        T.APPROVE_CHAIR_NAME,
                        T.IS_NEW_CUSTOMER,
                        T.USABLE_APP_CASH_LIMIT ,
                        T.USABLE_APP_NCASH_LIMIT ,
                        T.USABLE_APP_SUM_LIMIT ,
                        T.ALLOTMENT_CHANGE,

                        WHERE 1=1
                        AND T.STATE_NAME IN ('Aktif', 'Pasif','Transfer')

          ) LISTE

          AND LISTE.TASFIYE = 'Hayır'
          AND LISTE.TEKLIF_TÜRÜ <> 'Nakit Karşılıklı Teklif'
          AND LISTE.APPROVE_CHAIR NOT IN ('YKR','YKR1','KK','KK1','KKM')
          AND (LISTE.RISK_KRITERI_FLG = 1 OR LISTE.ILK_25_FLG = 1)
