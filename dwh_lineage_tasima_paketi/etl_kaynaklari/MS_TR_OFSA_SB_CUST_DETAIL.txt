           SELECT DISTINCT TARIH,MUS_NO,URUN_KODU,URUN,SUBE,ONCEKI_GUN_SUBE
             FROM(
                  SELECT A.AS_OF_DATE AS TARIH,
                         A.CUSTOMER_CODE AS MUS_NO,
                         A.ACCOUNT_NO AS HESAP_NO,
                         A.PRODUCT_TYPE AS URUN_KODU,
                         A.PRODUCT_NAME AS URUN,
                         A.CUSTOMER_BRANCH_CODE AS SUBE,

                AND A.CUSTOMER_BRANCH_CODE NOT IN (1,172,187,219)

                AND A.MAIN_GL_NO LIKE '1%'
                AND A.MAIN_GL_NO NOT LIKE '17%'
                AND NVL(A.PRODUCT_TYPE, '999') NOT IN ('0803002', '0903002', '0801003')

                     NVL(A.REDISCOUNT_103, 0) + NVL(A.REDISCOUNT_350059, 0) +
                     NVL(A.REDISCOUNT_35040, 0)) <> 0
               ORDER BY 1) 
               ) AST

      AND AST.MUS_NO=C.MUS_NO 
      AND AST.URUN_KODU=C.URUN_KODU 
      AND AST.SUBE<>AST.ONCEKI_GUN_SUBE
    GROUP BY

       C.URUN,
