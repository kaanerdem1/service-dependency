CREATE OR REPLACE PACKAGE BODY OFSA_SB.MS_EX_OFSA_SB_SEKERYATIRIM IS

PROCEDURE MS_EX_OFSA_SB_SEKERYATIRIM (SECIM NUMBER) IS

    ERROR_LOCATION     NUMBER;
    PID                NUMBER;
    PDATE              DATE;
    AYBASI             DATE;
    PASOFDATE          VARCHAR2(8);
    PAYBASI            VARCHAR(8);
    PCRITICAL_LVL      NUMBER := 1; --1:BLOCKER,2:CRITICAL,3:TRIVIAL,9:NO PROBLEM
    SEGM_TARIH         DATE;
    ONCEKIAYSONU       DATE;
    PONCEKIAYSONU      VARCHAR(8);
BEGIN

    ERROR_LOCATION := 1;

    SELECT PRM_VALUE, TO_DATE(PRM_VALUE, PRM_FORMAT),TO_DATE(MONTH_BEGIN, PRM_FORMAT),TO_DATE(LAST_SEGM_DATE, PRM_FORMAT), MONTH_BEGIN, last_day(add_months(TO_DATE(PRM_VALUE, PRM_FORMAT),-1) ), to_char(last_day(add_months(TO_DATE(PRM_VALUE, PRM_FORMAT),-1) ),'yyyymmdd')
      INTO PASOFDATE, PDATE,AYBASI,SEGM_TARIH, PAYBASI, ONCEKIAYSONU,PONCEKIAYSONU
      FROM OFSA_SB.SY_SYSTEM_PARAMETERS T
      WHERE PARAMETER = 'OFSA_SB_DATE';

     

      IF PDATE<=SEGM_TARIH THEN --SEGMENTASYON ATAMASI
      SEGM_TARIH:=SEGM_TARIH;
      ELSE
      SEGM_TARIH:=PDATE;
      END IF;

    PID := OFSA_SB.MN_UTILITY.INSERT_PLOG(PMODULE      => 'MS_EX_OFSA_SB_SEKERYATIRIM',
                                        PSUBMODULE   => 'EX_OFSA_SB_SEKERYATIRIM',
                                        PROJECT_NAME => 'OFSA_SB',
                                        PASOFDATE    => PDATE,
                                        PCRITICAL    => PCRITICAL_LVL);
   /*****************************************************************************************************************************/
    EXECUTE IMMEDIATE 'TRUNCATE TABLE OFSA_SB.OFSA_EX_SEKERYATIRIM';

    EXECUTE IMMEDIATE 'ALTER SESSION ENABLE PARALLEL DML';


    IF SECIM = 1 THEN
      

INSERT
      INTO OFSA_SB.OFSA_EX_SEKERYATIRIM

     (
          SELECT 
           PDATE AS AS_OF_DATE,
           NVL(TO_NUMBER(SKBNKMUSNO),999999999) AS MUS_NO,
           OFSA_SB.MN_UTILITY.MAP_OFSA_ISKOLU (BUSINESS_SEGMENT_CODE   => SEG.BUSINESS_SEGMENT_CODE) AS ISKOLU,
           SEG.BUSINESS_SEGMENT_CODE AS SEGMENT,
           CASE WHEN TO_NUMBER(I.AGENTID) IN ('6','220', '122','372') THEN -1 ELSE TO_NUMBER(ORG.PARENT_CODE) END BOLGE,
           TO_NUMBER(I.AGENTID) AS SUBE,
           NVL(TO_NUMBER(C.REPRESENTATIVE_FILE_NO),'999999') AS MT,
           '88888888' AS URUN_KODU,
           'SEKERYATIRIM KOMISYONLARI' AS URUN,
           'TL' AS PB,
           'SEKERYATIRIM_MUSTERI' AS KAYNAK,
           SUM(NVL(I.HSKOMISYONTUTARI,0)) *0.25 AS HS_KOMISYON_TUTARI, --Hisse Senedi: %25 Viop:  %25 Yatırım Fonu: %40 UW(Hisse senedi halka arz): %100 serkan hızlı 21.01.2020 mailine istinaden
           SUM(NVL(I.VIKOMISYONTUTARI,0)) *0.25 AS VOB_KOMISYON_TUTARI,
           0 AS FON_VERIM,
           SUM(NVL(I.UWKOMISYONTUTARI,0)) *1    AS HALKA_ARZ_KOMISYON_TUTARI,
           SUM(NVL(I.SGSGMKVERIM,0))      *0    AS SGMK_VERIM,
           SUM(NVL(I.SWKOMISYONTUTARI,0)) *0    AS SWEET_KOMISYON_TUTARI
  FROM     BI_APP.BATCHRPRTRSLT_CUSTEFFSUM_MNTLY  I 
           left join OLAP.FC_CUSTOMER C 
                  on C.AS_OF_DATE=PDATE 
                      AND to_number(NVL(SKBNKMUSNO,999999999))=to_number(C.CUSTOMER_CODE)
           left join OLAP.FC_CUSTOMER SEG
                  on SEG.AS_OF_DATE=SEGM_TARIH 
                      AND to_number(NVL(SKBNKMUSNO,999999999))=to_number(SEG.CUSTOMER_CODE)           
           left join OLAP.DM_ORGANIZATION ORG 
                  on TO_NUMBER(I.AGENTID)=TO_NUMBER(ORG.GL_ORG_CODE)
     WHERE I.ETL_DATE = PASOFDATE     
     GROUP BY 
           PDATE, 
           NVL(TO_NUMBER(SKBNKMUSNO),999999999),
           OFSA_SB.MN_UTILITY.MAP_OFSA_ISKOLU (BUSINESS_SEGMENT_CODE   => SEG.BUSINESS_SEGMENT_CODE),
           SEG.BUSINESS_SEGMENT_CODE,
           CASE WHEN TO_NUMBER(I.AGENTID) IN ('6','220', '122','372') THEN -1 ELSE TO_NUMBER(ORG.PARENT_CODE) END,
           TO_NUMBER(I.AGENTID),
           NVL(TO_NUMBER(C.REPRESENTATIVE_FILE_NO),'999999') 
     UNION ALL
     SELECT 
           PDATE AS AS_OF_DATE,
           NVL(TO_NUMBER(C.CUSTOMER_CODE),999999999) AS MUS_NO,
           OFSA_SB.MN_UTILITY.MAP_OFSA_ISKOLU (BUSINESS_SEGMENT_CODE   => C.SEGMENT) AS ISKOLU,
           C.SEGMENT AS SEGMENT,
           CASE WHEN TO_NUMBER(C.BRANCH_CODE) IN ('6','220', '122','372') THEN -1 ELSE TO_NUMBER(B.BOLGE_KODU /*B.REGION_CODE*/) END BOLGE,
           TO_NUMBER(C.BRANCH_CODE) AS SUBE,
           NVL(TO_NUMBER(C.OFF_CODE),'999999') AS MT,
           '88888888' AS URUN_KODU,
           'SEKERYATIRIM KOMISYONLARI' AS URUN,
           'TL' AS PB,
           'SEKERYATIRIM_MUSTERI' AS KAYNAK,
           0 AS HS_KOMISYON_TUTARI, --Hisse Senedi: %25 Viop:  %25 Yatırım Fonu: %40 UW(Hisse senedi halka arz): %100 serkan hızlı 21.01.2020 mailine istinaden
           0 AS VOB_KOMISYON_TUTARI,
           ROUND(SUM(A.TUTARTL*A.PAYLASIMORANI*A.YONETIMUCRETI*0.8),2)  AS FON_VERIM,
           0    AS HALKA_ARZ_KOMISYON_TUTARI,
           0    AS SGMK_VERIM,
           0    AS SWEET_KOMISYON_TUTARI 
      FROM BI_APP.V_BATCHREPORTRESULT_FUNDSTOCK@QADATA_LINK A,
           BI_APP.vw_dm_dim_customer_history C,
           OFSA_SB.BOLGELER B
     WHERE A.STOKTARIHI IN (PDATE, PDATE-1)        
       AND C.AS_OF_DATE = PDATE--A.STOKTARIHI 
       AND TO_NUMBER(A.MBBNO)=TO_NUMBER(C.CUSTOMER_CODE)
       AND C.BRANCH_CODE=B.SUBE_KODU --BRANCH_CODE
     GROUP BY 
           PDATE,
           NVL(TO_NUMBER(C.CUSTOMER_CODE),999999999),
           OFSA_SB.MN_UTILITY.MAP_OFSA_ISKOLU (BUSINESS_SEGMENT_CODE   => C.SEGMENT),
           C.SEGMENT,
           CASE WHEN TO_NUMBER(C.BRANCH_CODE) IN ('6','220', '122','372') THEN -1 ELSE TO_NUMBER(B.BOLGE_KODU/*REGION_CODE*/) END,
           TO_NUMBER(C.BRANCH_CODE),
           NVL(TO_NUMBER(C.OFF_CODE),'999999')
     );

      
      
      COMMIT;
      
 ELSE 
 
  INSERT
      INTO OFSA_SB.OFSA_EX_SEKERYATIRIM

     (
         SELECT 
           PDATE AS AS_OF_DATE,
           NVL(TO_NUMBER(C.CUSTOMER_CODE),999999999) AS MUS_NO,
           OFSA_SB.MN_UTILITY.MAP_OFSA_ISKOLU (BUSINESS_SEGMENT_CODE   => C.SEGMENT) AS ISKOLU,
           C.SEGMENT AS SEGMENT,
           CASE WHEN TO_NUMBER(C.BRANCH_CODE) IN ('6','220', '122','372') THEN -1 ELSE TO_NUMBER(B.BOLGE_KODU /*B.REGION_CODE*/) END BOLGE,
           TO_NUMBER(C.BRANCH_CODE) AS SUBE,
           NVL(TO_NUMBER(C.OFF_CODE),'999999') AS MT,
           '88888888' AS URUN_KODU,
           'SEKERYATIRIM KOMISYONLARI' AS URUN,
           'TL' AS PB,
           'SEKERYATIRIM_MUSTERI' AS KAYNAK,
           0 AS HS_KOMISYON_TUTARI, --Hisse Senedi: %25 Viop:  %25 Yatırım Fonu: %40 UW(Hisse senedi halka arz): %100 serkan hızlı 21.01.2020 mailine istinaden
           0 AS VOB_KOMISYON_TUTARI,
           ROUND(SUM(A.TUTARTL*A.PAYLASIMORANI*A.YONETIMUCRETI*0.8),2)  AS FON_VERIM,
           0    AS HALKA_ARZ_KOMISYON_TUTARI,
           0    AS SGMK_VERIM,
           0    AS SWEET_KOMISYON_TUTARI 
      FROM BI_APP.V_BATCHREPORTRESULT_FUNDSTOCK A,
           BI_APP.vw_dm_dim_customer_history C,
           OFSA_SB.BOLGELER B,
           OLAP.DM_TIME t
     WHERE A.STOKTARIHI = PDATE-1          
       AND C.AS_OF_DATE = PDATE-1
       AND TO_NUMBER(A.MBBNO)=TO_NUMBER(C.CUSTOMER_CODE)
       AND T.CALENDAR_DATE = A.STOKTARIHI
       AND (CASE WHEN T.CALENDAR_MONTH_END = T.CALENDAR_DATE THEN 1 ELSE 0 END) = 0
       AND C.BRANCH_CODE=B.SUBE_KODU --BRANCH_CODE
     GROUP BY 
           PDATE,
           NVL(TO_NUMBER(C.CUSTOMER_CODE),999999999),
           OFSA_SB.MN_UTILITY.MAP_OFSA_ISKOLU (BUSINESS_SEGMENT_CODE   => C.SEGMENT),
           C.SEGMENT,
           CASE WHEN TO_NUMBER(C.BRANCH_CODE) IN ('6','220', '122','372') THEN -1 ELSE TO_NUMBER(B.BOLGE_KODU/*REGION_CODE*/) END,
           TO_NUMBER(C.BRANCH_CODE),
           NVL(TO_NUMBER(C.OFF_CODE),'999999')
     );

 
    /*******************************************************************************************************************/
   
  END IF;    
   OFSA_SB.MN_UTILITY.UPDATE_PLOG(PID => PID, SUCCESS => 1, PCRITICAL => 9);

  EXCEPTION
    WHEN OTHERS THEN
      OFSA_SB.MN_UTILITY.UPDATE_PLOG(PID => PID,
                                   SUCCESS => 0,
                                   FAIL => 'POSITION: ' || ERROR_LOCATION || ' ERROR: ' || ABS(SQLCODE) || ':' || SQLERRM || ' ' || DBMS_UTILITY.FORMAT_ERROR_BACKTRACE,
                                   PCRITICAL => PCRITICAL_LVL);
      ROLLBACK;
      RAISE;
  END;

END;
