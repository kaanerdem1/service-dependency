CREATE OR REPLACE PACKAGE BODY OFSA_SB.MS_EX_OFSA_SB_FONLAMA_GETIRISI IS

PROCEDURE MS_EX_OFSA_SB_FONLAMA_GETIRISI (SECIM NUMBER) IS

    ERROR_LOCATION     NUMBER;
    PID                NUMBER;
    PDATE              DATE;
    AYBASI             DATE;
    PASOFDATE          VARCHAR2(8);
    PCRITICAL_LVL      NUMBER := 1; --1:BLOCKER,2:CRITICAL,3:TRIVIAL,9:NO PROBLEM
    SEGM_TARIH         DATE;
    READY              NUMBER;  
    YILBASI            DATE;  
    V_ORT_HAVUZ        NUMBER;
BEGIN

    ERROR_LOCATION := 1;

    SELECT PRM_VALUE, TO_DATE(PRM_VALUE, PRM_FORMAT),TO_DATE(MONTH_BEGIN, PRM_FORMAT),TO_DATE(LAST_SEGM_DATE, PRM_FORMAT),trunc(to_date(PRM_VALUE, 'yyyymmdd'), 'YYYY')
      INTO PASOFDATE, PDATE,AYBASI,SEGM_TARIH, YILBASI
      FROM OFSA_SB.SY_SYSTEM_PARAMETERS T
      WHERE PARAMETER = 'OFSA_SB_DATE';
      
    SELECT DISTINCT NVL(POOL_RATE, COST_RATE) 
    INTO V_ORT_HAVUZ
           FROM SUMMARY_PROD.GL_LIABILITY
          WHERE AS_OF_DATE = PDATE
            AND MODULE= 'VDZ'
            AND CURRENCY_CODE='TL';

      IF PDATE<=SEGM_TARIH THEN --SEGMENTASYON ATAMASI
      SEGM_TARIH:=SEGM_TARIH;
      ELSE
      SEGM_TARIH:=PDATE;
      END IF;

    PID := OFSA_SB.MN_UTILITY.INSERT_PLOG(PMODULE    => 'MS_EX_OFSA_SB_FONLAMA_GETIRISI',
                                        PSUBMODULE   => 'EX_OFSA_SB_FONLAMA_GETIRISI',
                                        PROJECT_NAME => 'OFSA_SB',
                                        PASOFDATE    => PDATE,
                                        PCRITICAL    => PCRITICAL_LVL);
   /*****************************************************************************************************************************/
    EXECUTE IMMEDIATE 'TRUNCATE TABLE OFSA_SB.OFSA_EX_FONLAMA_GETIRISI';

    EXECUTE IMMEDIATE 'ALTER SESSION ENABLE PARALLEL DML';

  --------------------------------------------------------------------
        LOOP
            SELECT COUNT(*) 
            INTO READY
            FROM SUMMARY_PROD.GL_LIABILITY
            WHERE AS_OF_DATE=PDATE
            AND COST_RATE IS NOT NULL
            AND MODULE='VDL';
            EXIT WHEN READY>75000;
            SLEEP(300000);--5DK
          END LOOP;
   -------------------------------------------------------------------
   
         LOOP
            SELECT COUNT(*) 
            INTO READY
            FROM SUMMARY_PROD.GL_LIABILITY
            WHERE AS_OF_DATE=PDATE
            AND COST_RATE IS NOT NULL
            AND MODULE='VDZ';
            EXIT WHEN READY>3720000;
            SLEEP(300000);--5DK
          END LOOP;
    -------------------------------------------------------------------
          
   IF SECIM=0 THEN --Eger 0 seçilirse günlük,0 dışında bir veri seçilirse aylık yükler
     AYBASI:=PDATE;
    END IF; 

    INSERT
      INTO OFSA_SB.OFSA_EX_FONLAMA_GETIRISI
  (
   SELECT
        MEVD.TARIH AS AS_OF_DATE,
        MEVD.MUS_NO AS MUS_NO,
        MEVD.SEGM AS SEGMENT,
       OFSA_SB.MN_UTILITY.MAP_OFSA_ISKOLU (BUSINESS_SEGMENT_CODE   => MEVD.SEGM) AS ISKOLU,
                        
        CASE WHEN TO_NUMBER(MEVD.SUBE) IN ('6','220','372','122') THEN -1 ELSE TO_NUMBER(ORG.PARENT_CODE) END BOLGE,
        TO_NUMBER(MEVD.SUBE) AS SUBE,
        NVL(MEVD.OFF_CODE, 999999) AS MT,
        NVL(MEVD.URUN_KODU,'88888888') AS URUN_KODU,
        NVL(MEVD.PRODUCT_NAME, 'DIGER_URUN') AS URUN,
        MEVD.CCY AS PB,
        'FONLAMA_GETIRISI' AS KAYNAK,
        NVL(ROUND(SUM(CASE WHEN MEVD.TUR='FONLAMA_GETIRI_MEVD_BLOKE' THEN (MEVD.ORT_BAKIYE*MEVD.ORT_HAVUZ*MEVD.GUN/36500) ELSE 0 END),2),0) AS MEVD_BLOKE_TUTAR,
        NVL(ROUND(SUM(CASE WHEN MEVD.TUR='FONLAMA_GETIRI_GM_BILANCO' THEN (MEVD.ORT_BAKIYE*MEVD.ORT_HAVUZ*MEVD.GUN/36500) ELSE 0 END),2),0) AS GM_BILANCO_TUTAR,
        0 AS KK_ALACAK_BKY        
      FROM OLAP.DM_ORGANIZATION ORG,
        (
        SELECT
          PDATE                    AS TARIH,
          CASE WHEN TO_NUMBER(A.BRANCH_CODE) IN ('188','1','219') THEN TO_NUMBER(C.MAIN_BRANCH_CODE) ELSE TO_NUMBER(A.BRANCH_CODE) END AS SUBE,
          CASE WHEN TO_NUMBER(A.BRANCH_CODE) IN ('188','1','219') THEN 'FONLAMA_GETIRI_GM_BILANCO' ELSE 'FONLAMA_GETIRI_MEVD_BLOKE' END TUR,
          A.CUSTOMER_CODE          AS MUS_NO,
          SEG.BUSINESS_SEGMENT_CODE  AS SEGM,
          C.REPRESENTATIVE_FILE_NO AS OFF_CODE,
          NVL(PRODUCT_TYPE,888888) URUN_KODU,
          A.PRODUCT_NAME,
          (TO_NUMBER(PDATE - AYBASI)+1) AS GUN,
          A.CURRENCY_CODE AS CCY,
          SUM((A.BS_MAIN_BALANCE) * nvl(a.cost_rate,A.POOL_RATE) ) / SUM((A.BS_MAIN_BALANCE)+1)    AS ORT_HAVUZ,
          SUM((A.BS_MAIN_BALANCE) / (TO_NUMBER(PDATE - AYBASI)+1)) AS ORT_BAKIYE
        FROM SUMMARY_PROD.GL_LIABILITY A, OLAP.FC_CUSTOMER C, OLAP.FC_CUSTOMER SEG
        WHERE A.AS_OF_DATE BETWEEN AYBASI AND PDATE  -- KONTROL EDILEN AYIN ILK VE SON GÜNÜ
          AND SUBSTR(A.MAIN_GL_NO,1,3) BETWEEN '300' AND '319'
          AND TO_NUMBER(A.BRANCH_CODE)||'-'||A.ACCOUNT_NO NOT IN (SELECT DISTINCT TO_NUMBER(A.MEVD_SUBE_KODU)||'-'||A.MEVD_HESAP_NO FROM OFSA_SB.TEMP_NAKIT_KARS_KRD A) --NAKIT KARŞILIKLI KREDININ TUTULDUĞU HESAP OLUP 27/04/2021 RE: IZMIR ŞUBE KARLILIK  SUBJ MAILE GÖRE ELENMIŞTIR,01.06.2021 TURKCOM TURIZM NKK HK  SBJ MAILE GÖRE ELENMIŞTIR
          AND A.CUSTOMER_CODE=C.CUSTOMER_CODE
          AND C.AS_OF_DATE=PDATE
          AND A.CUSTOMER_CODE=SEG.CUSTOMER_CODE
          AND SEG.AS_OF_DATE=SEGM_TARIH
        GROUP BY
          CASE WHEN TO_NUMBER(A.BRANCH_CODE) IN ('188','1','219') THEN TO_NUMBER(C.MAIN_BRANCH_CODE) ELSE TO_NUMBER(A.BRANCH_CODE) END,
          CASE WHEN TO_NUMBER(A.BRANCH_CODE) IN ('188','1','219') THEN 'FONLAMA_GETIRI_GM_BILANCO' ELSE 'FONLAMA_GETIRI_MEVD_BLOKE' END,
          A.CUSTOMER_CODE,
          SEG.BUSINESS_SEGMENT_CODE,
          C.REPRESENTATIVE_FILE_NO,
          NVL(PRODUCT_TYPE,888888),
          A.PRODUCT_NAME,
          A.CURRENCY_CODE
          ---HAVING SUM(A.BS_MAIN_BALANCE)+1<>0------- 3 kAYITTA ÇIKAN bÖLEN 0 HATASINI GİDERMEK İÇİN EKLENDİ 15.01.2024 Dilay          
          UNION ALL
       select 
       PDATE                    AS TARIH,
       CASE WHEN TO_NUMBER(o.org_code) IN ('188','1','219') THEN TO_NUMBER(C.MAIN_BRANCH_CODE) ELSE TO_NUMBER(o.org_code) END AS SUBE,
       CASE WHEN TO_NUMBER(o.org_code) IN ('188','1','219') THEN 'FONLAMA_GETIRI_GM_BILANCO' ELSE 'FONLAMA_GETIRI_MEVD_BLOKE' END TUR,
       to_char(o.cus_no)               AS MUS_NO, 
       SEG.BUSINESS_SEGMENT_CODE         AS SEGM,
       c.representative_file_no        AS OFF_CODE,
       '88888888'                      AS URUN_KODU,
       O.PRODUCT_TYPE                  AS PRODUCT_NAME,
       (TO_NUMBER(PDATE - AYBASI)+1)   AS GUN,
       O.ISO_CURRENCY_CD               AS CCY,
       o.reference_rate                AS ORT_HAVUZ,
       SUM(o.bal_ave) AS ORT_BAKIYE       
       from ofsa_daily.account@QADATA_LINK o, --ofsa_daily.account@ofsa_on_olaxrun o,
            olap.fc_customer c,
            olap.fc_customer SEG
       where o.as_of_date = PDATE
       and   c.as_of_date= PDATE
       and   o.cus_no=c.customer_code
       and   SEG.as_of_date= SEGM_TARIH
       and   o.cus_no=SEG.customer_code
       and   product_type='BLOKE' 
       group by  CASE WHEN TO_NUMBER(o.org_code) IN ('188','1','219') THEN TO_NUMBER(C.MAIN_BRANCH_CODE) ELSE TO_NUMBER(o.org_code) END,
       CASE WHEN TO_NUMBER(o.org_code) IN ('188','1','219') THEN 'FONLAMA_GETIRI_GM_BILANCO' ELSE 'FONLAMA_GETIRI_MEVD_BLOKE' END,
       o.cus_no, 
       SEG.BUSINESS_SEGMENT_CODE,
       c.representative_file_no,
       O.PRODUCT_TYPE,
       O.ISO_CURRENCY_CD,
       o.reference_rate 
          ) MEVD
      WHERE   TO_NUMBER(MEVD.SUBE)=TO_NUMBER(ORG.GL_ORG_CODE)
      GROUP BY MEVD.TARIH,
      OFSA_SB.MN_UTILITY.MAP_OFSA_ISKOLU (BUSINESS_SEGMENT_CODE   => MEVD.SEGM),
      CASE WHEN TO_NUMBER(MEVD.SUBE) IN ('6','220','372','122') THEN -1 ELSE TO_NUMBER(ORG.PARENT_CODE) END,
          TO_NUMBER(MEVD.SUBE),
          MEVD.OFF_CODE,
          MEVD.SEGM,
          MEVD.CCY,
          MEVD.URUN_KODU,
          MEVD.PRODUCT_NAME,
          MEVD.MUS_NO 
UNION ALL

        'TAKAS_CEKI'                AS URUN,
        'TL'                        AS PB,
        'FONLAMA_GETIRISI'          AS KAYNAK,

        0                                                                       AS GM_BILANCO_TUTAR,
        0                                                                       AS KK_ALACAK_BKY        
      FROM  
        (
         SELECT A.AS_OF_DATE AS PAY_DATE,
                DT.NEXT_WORKING_DAY,
                A.CUSTOMER_NO AS CUSTOMER_NO,
                A.BRANCH_CODE AS SUBE,
                CASE WHEN TO_NUMBER(A.BRANCH_CODE) IN ('6','220','372','122') THEN -1 ELSE TO_NUMBER(ORG.PARENT_CODE) END BOLGE,
                C.BUSINESS_SEGMENT_CODE,
                NVL(C.REPRESENTATIVE_FILE_NO, 999999) AS MT,
                A.OPERATION_DESC,
                A.PROCESS_DESCRIPTION,
                A.AMOUNT * -1 as AMOUNT
        FROM OLAP.FC_VOUCHER_DETAIL A,
             OLAP.FC_CUSTOMER C,
             OLAP.DM_ORGANIZATION ORG,
             OLAP.DM_TIME DT
       WHERE A.AS_OF_DATE = PDATE

       --  AND A.CONTRA_CUSTOMER_CODE=C.CUSTOMER_CODE
         AND A.CUSTOMER_NO=C.CUSTOMER_CODE
         AND TO_NUMBER(A.BRANCH_CODE) = ORG.GL_ORG_CODE
         AND A.AS_OF_DATE=DT.CALENDAR_DATE
         AND CONTRA_GL_CODE LIKE '392%' 
       --  AND CUSTOMER_NO='52980'
         AND PROCESS_ID in ('131126','131127')
        --AND CUSTOMER_NO='48463'
       -- AND CONTRA_CUSTOMER_CODE='36079044'
--          AND OPERATION_DESC IN
--      (
--        SELECT OPERATION_DESC
--          FROM OLAP.FC_VOUCHER_DETAIL A
--         WHERE A.AS_OF_DATE = to_date('20260522','yyyymmdd')
--           AND A.BRANCH_CODE='033'
--           AND A.PROCESS_ID='131126'
--           AND A.GL_CODE LIKE '392%'
--      )
        ) CEK
        where 1=1
     GROUP BY CEK.CUSTOMER_NO ,
              CEK.BUSINESS_SEGMENT_CODE ,
              OFSA_SB.MN_UTILITY.MAP_OFSA_ISKOLU (BUSINESS_SEGMENT_CODE   => CEK.BUSINESS_SEGMENT_CODE) ,                        
              CEK.BOLGE,
              TO_NUMBER(CEK.SUBE) ,
              CEK.MT                   
UNION ALL          
          SELECT
        MEVD.TARIH AS AS_OF_DATE,
        MEVD.MUS_NO AS MUS_NO,
        MEVD.SEGM AS SEGMENT,
        OFSA_SB.MN_UTILITY.MAP_OFSA_ISKOLU (BUSINESS_SEGMENT_CODE   => MEVD.SEGM) AS ISKOLU,
                        
        CASE WHEN TO_NUMBER(MEVD.SUBE) IN ('6','220','372','122') THEN -1 ELSE TO_NUMBER(ORG.PARENT_CODE) END BOLGE,
        TO_NUMBER(MEVD.SUBE) AS SUBE,
        NVL(MEVD.OFF_CODE, 999999) AS MT,
        NVL(MEVD.URUN_KODU,'88888888') AS URUN_KODU,
        NVL(MEVD.PRODUCT_NAME, 'DIGER_URUN') AS URUN,
        MEVD.CCY AS PB,
        'FNLMA-GTRSI-SILSUPUR' AS KAYNAK,
        NVL(ROUND(SUM(CASE WHEN MEVD.TUR='FONLAMA_GETIRI_MEVD_BLOKE' THEN (MEVD.ORT_BAKIYE*MEVD.ORT_HAVUZ*MEVD.BLOKE_GUN_SAYISI/36500) ELSE 0 END),2),0) AS MEVD_BLOKE_TUTAR,
        NVL(ROUND(SUM(CASE WHEN MEVD.TUR='FONLAMA_GETIRI_GM_BILANCO' THEN (MEVD.ORT_BAKIYE*MEVD.ORT_HAVUZ*MEVD.BLOKE_GUN_SAYISI/36500) ELSE 0 END),2),0) AS GM_BILANCO_TUTAR,
        0 AS KK_ALACAK_BKY        
      FROM OLAP.DM_ORGANIZATION ORG,
        (   SELECT --Sil süpür: bayi şubesine ekleme yapılıyor
                   PDATE AS TARIH,
                   CASE WHEN TO_NUMBER(A.BRANCH_CODE) IN ('188', '1', '219') THEN TO_NUMBER(C.MAIN_BRANCH_CODE) ELSE TO_NUMBER(A.BRANCH_CODE) END AS SUBE,
                   CASE WHEN TO_NUMBER(A.BRANCH_CODE) IN ('188', '1', '219') THEN 'FONLAMA_GETIRI_GM_BILANCO' ELSE 'FONLAMA_GETIRI_MEVD_BLOKE' END TUR,
                   TO_CHAR(B.BAYI_MUS_NO) AS MUS_NO,
                   SEG.BUSINESS_SEGMENT_CODE AS SEGM,
                   A.REPRESENTATIVE_FILE_NO AS OFF_CODE,
                   NVL(PRODUCT_TYPE, 888888) URUN_KODU,
                   A.PRODUCT_NAME,
                   (TO_NUMBER(PDATE - AYBASI) + 1) AS GUN,
                   A.CURRENCY_CODE AS CCY,
                   SUM((B.AMOUNT) * NVL(A.COST_RATE, A.POOL_RATE)) /
                   SUM((B.AMOUNT) + 1) AS ORT_HAVUZ,
                   SUM((B.AMOUNT) /(TO_NUMBER(PDATE - AYBASI) + 1)) AS ORT_BAKIYE,
                   BLOKE_GUN_SAYISI                   
               FROM SUMMARY_PROD.GL_LIABILITY A,
                    OLAP.FC_CUSTOMER C,
                    OLAP.FC_CUSTOMER SEG,
                    (SELECT  A.AS_OF_DATE,
                             T.BAYI_SUBE, --Şubeye ekle
                             T.BAYI_MUS_NO, --Müş noya eklenecek
                             T.BAYI_HESAP_NO, --joinle ve hesaba eklenecek
                             A.AMOUNT AS AMOUNT,
                             T.BLOKE_GUN_SAYISI
                        FROM OFSA_SB_ANA_ALT_FIRMA T, OLAP.FC_VOUCHER_DETAIL A
                       WHERE A.AS_OF_DATE BETWEEN AYBASI AND PDATE
                         AND T.ANA_FIRMA_MUS_NO=A.CUSTOMER_NO
                         AND A.PROCESS_ID = 150092
                         AND A.DIRECTION = 'A'
                         AND SUBSTR(A.OPERATION_DESC, 0,INSTR(A.OPERATION_DESC,' nolu')-1)= TO_CHAR(T.BAYI_HESAP_NO)) B                         
              WHERE A.AS_OF_DATE BETWEEN AYBASI AND PDATE
                AND SUBSTR(A.MAIN_GL_NO, 1, 3) BETWEEN '300' AND '319'
                AND (to_number(A.ACCOUNT_NO) = to_number(B.BAYI_HESAP_NO))
                AND A.CUSTOMER_CODE = B.BAYI_MUS_NO
                AND A.AS_OF_DATE = B.AS_OF_DATE
                AND A.CUSTOMER_CODE = C.CUSTOMER_CODE
                AND C.AS_OF_DATE = PDATE
                AND A.CUSTOMER_CODE = SEG.CUSTOMER_CODE
                AND SEG.AS_OF_DATE = SEGM_TARIH
              GROUP BY CASE WHEN TO_NUMBER(A.BRANCH_CODE) IN ('188', '1', '219') THEN TO_NUMBER(C.MAIN_BRANCH_CODE) ELSE TO_NUMBER(A.BRANCH_CODE) END,
                       CASE WHEN TO_NUMBER(A.BRANCH_CODE) IN ('188', '1', '219') THEN 'FONLAMA_GETIRI_GM_BILANCO'  ELSE 'FONLAMA_GETIRI_MEVD_BLOKE' END,
                       TO_CHAR(B.BAYI_MUS_NO),
                       SEG.BUSINESS_SEGMENT_CODE,
                       A.REPRESENTATIVE_FILE_NO,
                       NVL(PRODUCT_TYPE, 888888),
                       A.PRODUCT_NAME,
                       A.CURRENCY_CODE,BLOKE_GUN_SAYISI                       
            UNION ALL
             SELECT --Sil süpür: Ana şubesinden düşme yapılıyor

                    CASE WHEN TO_NUMBER(A.BRANCH_CODE) IN ('188', '1', '219') THEN TO_NUMBER(C.MAIN_BRANCH_CODE) ELSE TO_NUMBER(A.BRANCH_CODE) END AS SUBE,
                    CASE WHEN TO_NUMBER(A.BRANCH_CODE) IN ('188', '1', '219') THEN 'FONLAMA_GETIRI_GM_BILANCO' ELSE 'FONLAMA_GETIRI_MEVD_BLOKE' END TUR,

                               T.ANA_FIRMA_SUBE, ---şubeden düşülecek
                               T.ANA_FIRMA_MUS_NO, ---müs nodan düşülecek
                               T.ANA_FIRMA_HESAP_NO, ----(-)OLARAK ÇIKAR--joinle ve düş
                               A.AMOUNT * -1 AS AMOUNT,T.BLOKE_GUN_SAYISI
                          FROM OFSA_SB_ANA_ALT_FIRMA T, OLAP.FC_VOUCHER_DETAIL A
                         WHERE A.AS_OF_DATE BETWEEN AYBASI AND PDATE
                           AND T.ANA_FIRMA_MUS_NO=A.CUSTOMER_NO
                           AND A.PROCESS_ID = 150092
                           AND A.DIRECTION = 'A'
                           AND SUBSTR(A.OPERATION_DESC, 0,INSTR(A.OPERATION_DESC,' nolu')-1)= TO_CHAR(T.BAYI_HESAP_NO)) B
              WHERE A.AS_OF_DATE BETWEEN AYBASI AND PDATE
                AND SUBSTR(A.MAIN_GL_NO, 1, 3) BETWEEN '300' AND '319'
                AND (TO_NUMBER(A.ACCOUNT_NO) = to_number(B.ANA_FIRMA_HESAP_NO))
                AND A.CUSTOMER_CODE = B.ANA_FIRMA_MUS_NO
                AND A.AS_OF_DATE = B.AS_OF_DATE
                AND A.CUSTOMER_CODE = C.CUSTOMER_CODE
                AND C.AS_OF_DATE = PDATE
                AND A.CUSTOMER_CODE = SEG.CUSTOMER_CODE
                AND SEG.AS_OF_DATE = SEGM_TARIH
              GROUP BY CASE WHEN TO_NUMBER(A.BRANCH_CODE) IN ('188', '1', '219') THEN TO_NUMBER(C.MAIN_BRANCH_CODE) ELSE TO_NUMBER(A.BRANCH_CODE) END,
                       CASE WHEN TO_NUMBER(A.BRANCH_CODE) IN ('188', '1', '219') THEN 'FONLAMA_GETIRI_GM_BILANCO' ELSE'FONLAMA_GETIRI_MEVD_BLOKE' END,
                       TO_CHAR(B.ANA_FIRMA_MUS_NO),
                       SEG.BUSINESS_SEGMENT_CODE,
                       A.REPRESENTATIVE_FILE_NO,
                       NVL(PRODUCT_TYPE, 888888),
                       A.PRODUCT_NAME,
                       A.CURRENCY_CODE,BLOKE_GUN_SAYISI  
          )MEVD          
      WHERE TO_NUMBER(MEVD.SUBE)=TO_NUMBER(ORG.GL_ORG_CODE)    
      GROUP BY MEVD.TARIH,
      OFSA_SB.MN_UTILITY.MAP_OFSA_ISKOLU (BUSINESS_SEGMENT_CODE   => MEVD.SEGM),
      CASE WHEN TO_NUMBER(MEVD.SUBE) IN ('6','220','372','122') THEN -1 ELSE TO_NUMBER(ORG.PARENT_CODE) END,
          TO_NUMBER(MEVD.SUBE),
          MEVD.OFF_CODE,
          MEVD.SEGM,
          MEVD.CCY,
          MEVD.URUN_KODU,
          MEVD.PRODUCT_NAME,
          MEVD.MUS_NO          
    UNION ALL          
SELECT --Kredi kart alacak bakiyelerinden oluşan gelir için eklendi. Talep No: 91990 
        PDATE AS AS_OF_DATE,
        A.CUSTOMER_CODE AS MUS_NO,
        SEG.BUSINESS_SEGMENT_CODE AS SEGMENT,
        OFSA_SB.MN_UTILITY.MAP_OFSA_ISKOLU (BUSINESS_SEGMENT_CODE   => SEG.BUSINESS_SEGMENT_CODE) AS ISKOLU,                        
        CASE WHEN TO_NUMBER(CASE WHEN TO_NUMBER(A.ORG_CODE) IN ('188','1','219') THEN C.MAIN_BRANCH_CODE ELSE A.ORG_CODE END) IN ('6','220', '122','372') THEN -1 ELSE TO_NUMBER(ORG.PARENT_CODE) END BOLGE,
        CASE WHEN TO_NUMBER(A.ORG_CODE) IN ('188','1','219') THEN TO_NUMBER(C.MAIN_BRANCH_CODE) ELSE TO_NUMBER(A.ORG_CODE) END AS SUBE,
        NVL(C.REPRESENTATIVE_FILE_NO,999999) AS MT,
        A.PRODUCT_TYPE AS URUN_KODU,
        'KART ALACAK BAKIYESI' AS URUN,
        A.CURRENCY_CODE AS PB,
        'KART_ALACAK_BAKIYE' AS KAYNAK,
        0 AS MEVD_BLOKE_TUTAR,
        0 AS GM_BILANCO_TUTAR,
        KK_BAKIYE AS KK_ALACAK_BKY         
   FROM OLAP.DM_ORGANIZATION ORG,
        OLAP.FC_CUSTOMER  C,
        OLAP.FC_CUSTOMER SEG,
        (
        SELECT  A.AS_OF_DATE,
                TO_CHAR(A.MUSTERI_NO) AS CUSTOMER_CODE,
                A.BRANCH_CODE AS ORG_CODE,
                TO_CHAR(A.SOURCE_SYSTEM_PRODUCT_CODE) AS PRODUCT_TYPE,
                'TL' AS CURRENCY_CODE,
                ((SUM(NVL(AVAILABLE_AMOUNT,0))-SUM(NVL(MUSTERI_LIMIT,0)))*NVL(H.AVE_RATE,0))/365 AS KK_BAKIYE,
                SUM(NVL(A.AVAILABLE_AMOUNT,0)-NVL(A.MUSTERI_LIMIT,0)),
                H.AVE_RATE
           FROM OFSA_DAILY.CREDIT_CARDS@QADATA_LINK A,
                (SELECT AVG(NVL(COST_RATE,POOL_RATE))/100 AS AVE_RATE                                
                   FROM SUMMARY_PROD.GL_LIABILITY C
                  WHERE 1=1
                    AND AS_OF_DATE BETWEEN AYBASI AND PDATE
                    AND C.MODULE='VDZ'
                    AND C.CURRENCY_CODE='TL') H
                WHERE 1=1
                  AND A.AS_OF_DATE = PDATE
                  AND NVL(A.AVAILABLE_AMOUNT,0)>NVL(A.MUSTERI_LIMIT,0)
                  AND NVL(A.KART_BAKIYE,0)<0
            --and a.customer_code='35836241'   
            GROUP BY
                  A.AS_OF_DATE,
                  A.MUSTERI_NO,
                  A.BRANCH_CODE , 
                  TO_CHAR(A.SOURCE_SYSTEM_PRODUCT_CODE),
                  H.AVE_RATE            
                ) A         
        WHERE CASE WHEN TO_NUMBER(A.ORG_CODE) IN ('188','1','219') THEN TO_NUMBER(C.MAIN_BRANCH_CODE) ELSE TO_NUMBER(A.ORG_CODE) END=TO_NUMBER(ORG.GL_ORG_CODE)
             AND C.CUSTOMER_CODE=A.CUSTOMER_CODE
             AND C.AS_OF_DATE= PDATE
             AND SEG.CUSTOMER_CODE=A.CUSTOMER_CODE
             AND SEG.AS_OF_DATE= PDATE
             and ORG.STATE=1
  );


      COMMIT;

    /*******************************************************************************************************************/

   OFSA_SB.MN_UTILITY.UPDATE_PLOG(PID => PID, SUCCESS => 1, PCRITICAL => 9);

  EXCEPTION
    WHEN OTHERS THEN
      OFSA_SB.MN_UTILITY.UPDATE_PLOG(PID => PID,
                                   SUCCESS => 0,
                                   FAIL => 'POSITION: ' || ERROR_LOCATION || ' ERROR: ' || ABS(SQLCODE) || ':' || SQLERRM || ' ' || DBMS_UTILITY.FORMAT_ERROR_BACKTRACE,
                                   PCRITICAL => PCRITICAL_LVL);
      ROLLBACK;
      RAISE;
  END;

END;
