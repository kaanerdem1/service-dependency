CREATE OR REPLACE PACKAGE BODY OFSA_SB.MS_EX_OFSA_SB_MANUEL_KALEMLER IS

PROCEDURE MS_EX_OFSA_SB_MANUEL_KALEMLER /*(SECIM NUMBER) */ IS

    ERROR_LOCATION     NUMBER;
    PID                NUMBER;
    PDATE              DATE;
    AYBASI             DATE;
    PASOFDATE          VARCHAR2(8);
    PCRITICAL_LVL      NUMBER := 1; --1:BLOCKER,2:CRITICAL,3:TRIVIAL,9:NO PROBLEM
    SEGM_TARIH         DATE;
    AYSONU             DATE;
    SAYAC              NUMBER :=0;
    KART_CEK_SAYAC     NUMBER :=0;
    ONCEKIAYSONU       DATE;
    ONCEKIAYBASI       DATE;
BEGIN

    ERROR_LOCATION := 1;

      SELECT PRM_VALUE, 
      TO_DATE(PRM_VALUE, PRM_FORMAT),
      TO_DATE(MONTH_BEGIN, PRM_FORMAT),
      TO_DATE(LAST_SEGM_DATE, PRM_FORMAT),
      TO_DATE(TO_CHAR(TRUNC(TO_DATE(PRM_VALUE,'YYYYMMDD'),'MM')-1,'YYYYMMDD'),'YYYYMMDD'),
      trunc(add_months(TO_DATE(PRM_VALUE, PRM_FORMAT),-1),'MM')
      INTO PASOFDATE, PDATE,AYBASI,SEGM_TARIH,ONCEKIAYSONU, ONCEKIAYBASI
      FROM OFSA_SB.SY_SYSTEM_PARAMETERS T
      WHERE PARAMETER = 'OFSA_SB_DATE';
      -------------------------------------------------------
      SELECT A.CALENDAR_MONTH_END --Aysonu tarihi ataması yapılıyor.
      INTO AYSONU      
      FROM OLAP.DM_TIME A
      WHERE A.CALENDAR_DATE=PDATE;
      -------------------------------------------------------
      SELECT COUNT(1) --İlgili tarihte kayıt olma durumu knt ediliyor.
      INTO SAYAC
      FROM OFSA_SB.OFSA_TEMP_MEVD_PRIM_2021 B
      WHERE B.AS_OF_DATE=PDATE;
      -------------------------------------------------------

      SELECT COUNT(SINIF)
      INTO KART_CEK_SAYAC
      FROM OFSA_SB.OFSA_MANUEL_UPLOAD_LOG
      WHERE (SINIF='ŞUBEDE_BEKLEYEN_CEK_KARNE' or SINIF='SUBEDE_BEKLEYEN_TIC_KARTLAR')
      AND AS_OF_DATE=PDATE;
    
      -------------------------------------------------------

      IF PDATE<=SEGM_TARIH THEN --SEGMENTASYON ATAMASI
      SEGM_TARIH:=SEGM_TARIH;
      ELSE
      SEGM_TARIH:=PDATE;
      END IF;

    PID := OFSA_SB.MN_UTILITY.INSERT_PLOG(PMODULE      => 'MS_EX_OFSA_SB_MANUEL_KALEMLER',
                                        PSUBMODULE   => 'EX_OFSA_SB_MANUEL_KALEMLER',
                                        PROJECT_NAME => 'OFSA_SB',
                                        PASOFDATE    => PDATE,
                                        PCRITICAL    => PCRITICAL_LVL);
   
   
    /*******************************************************************************************************************/
    
/*    pdate:=to_date('20210331','yyyymmdd') ;
    AYSONU:=to_date('20210331','yyyymmdd') ;
    ONCEKIAYSONU:=to_date('20210228','yyyymmdd') ;*/
    
    --Vadeli ve Vadesiz mevduat primleri için ara tablo dolduruluyor
    
    EXECUTE IMMEDIATE 'TRUNCATE TABLE OFSA_SB.OFSA_EX_MANUEL_KALEMLER';
    EXECUTE IMMEDIATE 'ALTER session set nls_date_format= '||'yyyymmdd'||''; 
      /*IF SECIM = 1 THEN*/
      IF PDATE=AYSONU THEN

         DELETE OFSA_SB.OFSA_TEMP_MEVD_PRIM_2021   T
         WHERE T.AS_OF_DATE=PDATE;
         COMMIT;
    
      INSERT INTO OFSA_SB.OFSA_TEMP_MEVD_PRIM_2021  
      (
      
      SELECT 
          PDATE  AS AS_OF_DATE,
          A.sube,
          A.SEGM,
          GREATEST(SUM(CASE WHEN  A.AS_OF_DATE=PDATE  AND A.SINIF='Vadesiz' THEN A.TUTAR ELSE 0 END)-SUM(CASE WHEN A.AS_OF_DATE=ONCEKIAYSONU  AND A.SINIF='Vadesiz' THEN A.TUTAR ELSE 0 END),0) AS Gelisim_Vadesiz,         
          sum(CASE WHEN A.AS_OF_DATE= PDATE AND A.SINIF='Vadeli' THEN A.TUTAR ELSE 0 END) AS Stok,
          GREATEST(SUM(CASE WHEN  A.AS_OF_DATE= PDATE AND A.SINIF='Vadeli' AND a.ALT_SINIF='0-250K' THEN A.TUTAR ELSE 0 END)-SUM(CASE WHEN A.AS_OF_DATE= ONCEKIAYSONU  AND A.SINIF='Vadeli' AND a.ALT_SINIF='0-250K' THEN A.TUTAR ELSE 0 END),0) AS Gelisim_0_250k,
          GREATEST(SUM(CASE WHEN  A.AS_OF_DATE= PDATE AND A.SINIF='Vadeli' AND A.ALT_SINIF='250K-1M' THEN A.TUTAR ELSE 0 END)-SUM(CASE WHEN  A.AS_OF_DATE= ONCEKIAYSONU  AND A.SINIF='Vadeli' AND a.ALT_SINIF='250K-1M' THEN A.TUTAR ELSE 0 END),0) AS GELISIM_250K_1M,
          GREATEST(SUM(CASE WHEN  A.AS_OF_DATE= PDATE AND A.SINIF='Vadeli' AND A.ALT_SINIF='1M+' THEN A.TUTAR ELSE 0 END)-SUM(CASE WHEN  A.AS_OF_DATE= ONCEKIAYSONU AND A.SINIF='Vadeli' AND a.ALT_SINIF='1M+' THEN A.TUTAR ELSE 0 END),0) AS GELISIM_1M_UZERI,
          'TL' as PB
         FROM
(

select 
        PDATE  as_of_date, 
        a.customer_code as mus_no,
        'Vadeli' as sinif,
        CASE WHEN sum(SUM(a.main_balance)/( to_number((PDATE-ONCEKIAYSONU)))) over (PARTITION BY A.CUSTOMER_CODE,to_number(a.branch_code))<250000   THEN '0-250K'
             WHEN sum(SUM(a.main_balance)/( to_number((PDATE-ONCEKIAYSONU)))) over (PARTITION BY A.CUSTOMER_CODE,to_number(a.branch_code)) between 250000.01 and 1050000  THEN '250K-1M' 
             WHEN sum(SUM(a.main_balance)/( to_number((PDATE-ONCEKIAYSONU)))) over (PARTITION BY A.CUSTOMER_CODE,to_number(a.branch_code)) > 1050000  THEN '1M+' ELSE '0' END as alt_sinif,     
        to_number(a.branch_code) as sube,   
        o.org_name as sube_adi,
        case when to_number(case when a.branch_code = '188' then c.main_branch_code else a.branch_code end) in ('6','220','372','122' ) then '794' else o.parent_code end as bolge_kodu,
        case when to_number(case when a.branch_code = '188' then c.main_branch_code else a.branch_code end) in ('6','220','372','122' ) then 'KURUMSAL ŞUBELER' else o.parent_name end as bolge,         
        SEG.profit_center_code as segm,
        a.currency_code  as ccy, 
        sum(nvl(a.main_balance,0))/( to_number((PDATE-ONCEKIAYSONU))) as Tutar,
        sum(case when a.as_of_date= PDATE then (nvl(a.pool_rate,0) - nvl(a.interest_rate,0))*nvl(a.main_balance,0) /100 else 0 end) as SPREAD
    from summary_prod.gl_liability a, olap.fc_customer c, olap.fc_customer SEG, olap.dm_organization o  
    where a.as_of_date between ONCEKIAYSONU and PDATE
    and c.as_of_date = PDATE
    and a.customer_code = c.customer_code
    and SEG.as_of_date = SEGM_TARIH
    and a.customer_code = SEG.customer_code
    and  to_number(case when a.branch_code = '188' then c.main_branch_code else a.branch_code end) = to_number(o.gl_org_code)
    and substr(a.main_gl_no,1,3) between '310' and '325'
    and a.branch_code != '219'
    and nvl(a.main_balance,0) <> 0
    and a.currency_code = 'TL'
group by
       a.customer_code,a.branch_code, o.org_name,
       case when to_number(case when a.branch_code = '188' then c.main_branch_code else a.branch_code end) in ('6','220','372','122' ) then '794' else o.parent_code end,
       case when to_number(case when a.branch_code = '188' then c.main_branch_code else a.branch_code end) in ('6','220','372','122' ) then 'KURUMSAL ŞUBELER' else o.parent_name end,         
       SEG.profit_center_code,
       a.currency_code 
 UNION ALL 
 select 
        ONCEKIAYSONU  as_of_date, 
        a.customer_code as mus_no,
        'Vadeli' as sinif,
        CASE WHEN sum(SUM(a.main_balance)/30) over (PARTITION BY A.CUSTOMER_CODE,to_number(a.branch_code))<250000   THEN '0-250K'
             WHEN sum(SUM(a.main_balance)/30) over (PARTITION BY A.CUSTOMER_CODE,to_number(a.branch_code)) between 250000.01 and 1050000  THEN '250K-1M' 
             WHEN sum(SUM(a.main_balance)/30) over (PARTITION BY A.CUSTOMER_CODE,to_number(a.branch_code)) between 250000.01 and 1050000  THEN '1M+' ELSE '0' END as alt_sinif,     
        to_number(a.branch_code) as sube,   
        o.org_name as sube_adi,
        case when to_number(case when a.branch_code = '188' then c.main_branch_code else a.branch_code end) in ('6','220','372','122' ) then '794' else o.parent_code end as bolge_kodu,
        case when to_number(case when a.branch_code = '188' then c.main_branch_code else a.branch_code end) in ('6','220','372','122' ) then 'KURUMSAL ŞUBELER' else o.parent_name end as bolge,         
        SEG.profit_center_code as segm,
        a.currency_code  as ccy, 
        sum(nvl(a.main_balance,0))/ 30 as Tutar,
        sum(case when a.as_of_date= ONCEKIAYSONU then (nvl(a.pool_rate,0) - nvl(a.interest_rate,0))*nvl(a.main_balance,0) /100 else 0 end) as SPREAD
    from summary_prod.gl_liability a, olap.fc_customer c, olap.fc_customer SEG, olap.dm_organization o  
    where a.as_of_date between ONCEKIAYSONU-30 and ONCEKIAYSONU
    and c.as_of_date = PDATE
    and a.customer_code = c.customer_code
    and SEG.as_of_date = SEGM_TARIH
    and a.customer_code = SEG.customer_code
    and  to_number(case when a.branch_code = '188' then c.main_branch_code else a.branch_code end) = to_number(o.gl_org_code)
    and substr(a.main_gl_no,1,3) between '310' and '325'
    and a.branch_code != '219'
    and nvl(a.main_balance,0) <> 0
    and a.currency_code = 'TL'
group by
       a.customer_code,a.branch_code, o.org_name,
       case when to_number(case when a.branch_code = '188' then c.main_branch_code else a.branch_code end) in ('6','220','372','122' ) then '794' else o.parent_code end,
       case when to_number(case when a.branch_code = '188' then c.main_branch_code else a.branch_code end) in ('6','220','372','122' ) then 'KURUMSAL ŞUBELER' else o.parent_name end,         
       SEG.profit_center_code,
       a.currency_code                       
 UNION ALL
select 
        PDATE AS as_of_date, 
        a.customer_code as mus_no,
        'Vadesiz' as sinif,
        CASE WHEN sum(SUM(a.main_balance)/( to_number((PDATE-ONCEKIAYSONU)))) over (PARTITION BY A.CUSTOMER_CODE,to_number(a.branch_code))<250000   THEN '0-250K'
             WHEN sum(SUM(a.main_balance)/( to_number((PDATE-ONCEKIAYSONU)))) over (PARTITION BY A.CUSTOMER_CODE,to_number(a.branch_code)) between 250000.01 and 1050000  THEN '250K-1M' 
             WHEN sum(SUM(a.main_balance)/( to_number((PDATE-ONCEKIAYSONU)))) over (PARTITION BY A.CUSTOMER_CODE,to_number(a.branch_code)) between 250000.01 and 1050000  THEN '1M+' ELSE '0' END as alt_sinif,     
        to_number(a.branch_code) as sube,   
        o.org_name as sube_adi,
        case when to_number(case when a.branch_code = '188' then c.main_branch_code else a.branch_code end) in ('6','220','372','122' ) then '794' else o.parent_code end as bolge_kodu,
        case when to_number(case when a.branch_code = '188' then c.main_branch_code else a.branch_code end) in ('6','220','372','122' ) then 'KURUMSAL ŞUBELER' else o.parent_name end as bolge,         
        SEG.profit_center_code  as segm,
        a.currency_code, 
      sum(nvl(a.main_balance,0)/( to_number((PDATE-ONCEKIAYSONU))) ) as Tutar,
        0 as SPREAD
    from summary_prod.gl_liability a, olap.fc_customer c, olap.fc_customer SEG, olap.dm_organization o  
    where a.as_of_date between ONCEKIAYSONU and PDATE
    and c.as_of_date = PDATE
    and a.customer_code = c.customer_code
    and SEG.as_of_date = SEGM_TARIH
    and a.customer_code = SEG.customer_code
    and to_number(case when a.branch_code = '188' then c.main_branch_code else a.branch_code end) = to_number(o.gl_org_code)
    and substr(a.main_gl_no,1,3) between '300' and '309'
    and a.branch_code != '219'
    and nvl(a.main_balance,0) <> 0
group by
       a.customer_code,a.branch_code, o.org_name,
       case when to_number(case when a.branch_code = '188' then c.main_branch_code else a.branch_code end) in ('6','220','372','122' ) then '794' else o.parent_code end,
       case when to_number(case when a.branch_code = '188' then c.main_branch_code else a.branch_code end) in ('6','220','372','122' ) then 'KURUMSAL ŞUBELER' else o.parent_name end,         
       SEG.profit_center_code,
       a.currency_code 
 UNION ALL
select 
        PDATE  as_of_date, 
        a.customer_code as mus_no,
        'Vadeli' as sinif,
        CASE WHEN sum(SUM(a.main_balance_org*kur.EVALUATION_BUY_RATE)/( to_number((PDATE-ONCEKIAYSONU)))) over (PARTITION BY A.CUSTOMER_CODE,to_number(a.branch_code))<250000   THEN '0-250K'
             WHEN sum(SUM(a.main_balance_org*kur.EVALUATION_BUY_RATE)/( to_number((PDATE-ONCEKIAYSONU)))) over (PARTITION BY A.CUSTOMER_CODE,to_number(a.branch_code)) between 250000.01 and 1050000   THEN '250K-1M'
             WHEN sum(SUM(a.main_balance_org*kur.EVALUATION_BUY_RATE)/( to_number((PDATE-ONCEKIAYSONU)))) over (PARTITION BY A.CUSTOMER_CODE,to_number(a.branch_code))> 1050000.01  THEN '1M+' ELSE '0' END as alt_sinif,     
        to_number(a.branch_code) as sube,   
        o.org_name as sube_adi,
        case when to_number(case when a.branch_code = '188' then c.main_branch_code else a.branch_code end) in ('6','220','372','122' ) then '794' else o.parent_code end as bolge_kodu,
        case when to_number(case when a.branch_code = '188' then c.main_branch_code else a.branch_code end) in ('6','220','372','122' ) then 'KURUMSAL ŞUBELER' else o.parent_name end as bolge,         
        SEG.profit_center_code as segm,
        a.currency_code as ccy, 
        sum(nvl(a.main_balance_org,0)*kur.EVALUATION_BUY_RATE) / ( to_number((PDATE-ONCEKIAYSONU))) as Tutar,
        sum(case when a.as_of_date=PDATE then (nvl(a.pool_rate,0) - nvl(a.interest_rate,0))*nvl(a.main_balance,0) /100 else 0 end) as SPREAD
    from summary_prod.gl_liability a, olap.fc_customer c, olap.fc_customer SEG, olap.dm_organization o,  olap.fc_evaluation_rates kur  
    where a.as_of_date between ONCEKIAYSONU and PDATE
    and c.as_of_date = PDATE
    and kur.as_of_date= PDATE
    and a.customer_code = c.customer_code
    and a.customer_code = SEG.customer_code
    and SEG.as_of_date = SEGM_TARIH
    and  kur.CURRENCY_CODE=a.currency_code
    and to_number(case when a.branch_code = '188' then c.main_branch_code else a.branch_code end) = to_number(o.gl_org_code)
    and substr(a.main_gl_no,1,3) between '310' and '325'
    and a.branch_code != '219'
    and nvl(a.main_balance,0) <> 0
    and a.currency_code <> 'TL'  
group by
       a.customer_code,a.branch_code, o.org_name,
       case when to_number(case when a.branch_code = '188' then c.main_branch_code else a.branch_code end) in ('6','220','372','122' ) then '794' else o.parent_code end,
       case when to_number(case when a.branch_code = '188' then c.main_branch_code else a.branch_code end) in ('6','220','372','122' ) then 'KURUMSAL ŞUBELER' else o.parent_name end,         
       SEG.profit_center_code,
       a.currency_code    
 UNION ALL
select 
        ONCEKIAYSONU  as_of_date, 
        a.customer_code as mus_no,
        'Vadeli' as sinif,
        CASE WHEN sum(SUM(a.main_balance_org*kur.EVALUATION_BUY_RATE)/30) over (PARTITION BY A.CUSTOMER_CODE,to_number(a.branch_code))<250000   THEN '0-250K'
             WHEN sum(SUM(a.main_balance_org*kur.EVALUATION_BUY_RATE)/30) over (PARTITION BY A.CUSTOMER_CODE,to_number(a.branch_code)) between 250000.01 and 1050000   THEN '250K-1M'
             WHEN sum(SUM(a.main_balance_org*kur.EVALUATION_BUY_RATE)/30) over (PARTITION BY A.CUSTOMER_CODE,to_number(a.branch_code))> 1050000.01  THEN '1M+' ELSE '0' END as alt_sinif,     
        to_number(a.branch_code) as sube,   
        o.org_name as sube_adi,
        case when to_number(case when a.branch_code = '188' then c.main_branch_code else a.branch_code end) in ('6','220','372','122' ) then '794' else o.parent_code end as bolge_kodu,
        case when to_number(case when a.branch_code = '188' then c.main_branch_code else a.branch_code end) in ('6','220','372','122' ) then 'KURUMSAL ŞUBELER' else o.parent_name end as bolge,         
        SEG.profit_center_code as segm,
        a.currency_code as ccy, 
        sum(nvl(a.main_balance_org,0)*kur.EVALUATION_BUY_RATE) / 30 as Tutar,
        sum(case when a.as_of_date = ONCEKIAYSONU then (nvl(a.pool_rate,0) - nvl(a.interest_rate,0))*nvl(a.main_balance,0) /100 else 0 end) as SPREAD
    from summary_prod.gl_liability a, olap.fc_customer c, olap.fc_customer SEG, olap.dm_organization o,  olap.fc_evaluation_rates kur  
    where a.as_of_date between ONCEKIAYSONU-30 and ONCEKIAYSONU
    and c.as_of_date = PDATE
    and kur.as_of_date= PDATE
    and a.customer_code = c.customer_code
    and SEG.as_of_date = SEGM_TARIH
    and a.customer_code = SEG.customer_code
    and  kur.CURRENCY_CODE=a.currency_code
    and to_number(case when a.branch_code = '188' then c.main_branch_code else a.branch_code end) = to_number(o.gl_org_code)
    and substr(a.main_gl_no,1,3) between '310' and '325'
    and a.branch_code != '219'
    and nvl(a.main_balance,0) <> 0
    and a.currency_code <> 'TL'  
group by
       a.customer_code,a.branch_code, o.org_name,
       case when to_number(case when a.branch_code = '188' then c.main_branch_code else a.branch_code end) in ('6','220','372','122' ) then '794' else o.parent_code end,
       case when to_number(case when a.branch_code = '188' then c.main_branch_code else a.branch_code end) in ('6','220','372','122' ) then 'KURUMSAL ŞUBELER' else o.parent_name end,         
       SEG.profit_center_code,
       a.currency_code       
 )  a   
 group by PDATE,
          A.sube,
          A.SEGM              
          );
          COMMIT;
    
   
    --Vadeli mevduat primi ekleniyor
    
    --*******30.06.2021 TARİHİ İTİBARIYLA YENİ KURALLAR UYGULAMAYA ALINMIŞTIR. A.Y. **********---------
     
 --buradan başla    
    INSERT
      INTO OFSA_SB.OFSA_EX_MANUEL_KALEMLER
     (            
     -----ŞUBE TİPİ KURUMSAL VEYA TİCARİ AĞIRLIKLI OLMAYAN ŞUBELER İÇİN VADELİ MEVDUAT PRİMİ--------  
     
     /*Kriterler: vadesiz mevduat gelişimi 0'dan büyük ve TYM H/G %90 ve üzeri olan şubelere 
       0-250k TL gelişim için %5, 250-1050K TL gelişim için %2.5 prim verilir. YP mevduata prim verilmez. 30.06.2021 A.Y.*/
         
     SELECT 
           PDATE AS AS_OF_DATE,
           999999999 MUS_NO,
           '99-SUBE_HAVUZ' AS ISKOLU,
           '99998' SEGMENT,
           CASE WHEN TO_NUMBER(Z.SUBE) IN ('6','122','220', '372') THEN -1 ELSE TO_NUMBER(ORG.PARENT_CODE) END BOLGE,
           Z.sube AS SUBE,
           '999999' AS MT,
           '88888888' AS URUN_KODU,
           'Vadeli Mevduat Primi' AS URUN,
           'TL' AS PB,
           'MANUEL_KALEMLER' AS KAYNAK,     
           /*ROUND(SUM(Z.GELISIMDEN_PRIM250k * VADESIZ_CARPAN * TYM_CARPAN ) ,2 ) VADELI_MEVD_PRIM_250,
           ROUND(SUM(Z.GELISIMDENPRIM250_1050k * VADESIZ_CARPAN * TYM_CARPAN ) ,2 ) VADELI_MEVD_PRIM_1050,*/
                ROUND(SUM(Z.GELISIMDEN_PRIM250k * VADESIZ_CARPAN * TYM_CARPAN ),2) + 
                ROUND(SUM(Z.GELISIMDENPRIM250_1050k * VADESIZ_CARPAN * TYM_CARPAN ),2 ) BAKIYE                                                                                                                                                      
     FROM OLAP.DM_ORGANIZATION ORG,
     
     (    SELECT V.sube,
                 V.segm,
                 V.SUBE_TIPI,
                 V.GELISIMDEN_PRIM250k AS GELISIMDEN_PRIM250k,
                 V.GELISIMDENPRIM250_1050k AS GELISIMDENPRIM250_1050k,
                 V.GELISIMDENPRIM1m AS GELISIMDENPRIM1m,
                 CASE WHEN V.VADESIZ_GELISIM >0 THEN 1 ELSE 0 END AS VADESIZ_CARPAN, --VADESİZ ORTALAMASI AZALMIŞ İSE CARPAN 0 OLUR VE PRİM ALAMAZ  A.Y.
                 TYM.TYM AS TYM_CARPAN --TYM HG %90 ALTI İSE CARPAN 0 OLUR VE PRİM ALAMAZ  A.Y.
                 
          FROM 
          (                  
           SELECT to_number(g.SUBE) as sube,
                  g.SEGM,
                  SUBE_TIPI.SUBE_TIPI as SUBE_TIPI,
                  round(sum(g.GELISIM_VADESIZ),0)  AS VADESIZ_GELISIM,
                  round(sum(g.GELISIM_0_250K),0) GELISIM_0_250K,
                  round(sum(g.GELISIM_250K_1M),0) GELISIM_250_1050K,
                  round(sum(g.GELISIM_250K_1M),0) GELISIM_1M_Uzeri,
                  round(sum(g.GELISIM_0_250K) * 5 / 1200,0) AS GELISIMDEN_PRIM250k,
                  round(sum(g.GELISIM_250K_1M) * 2.5 / 1200,0) AS GELISIMDENPRIM250_1050k , 
                  round(sum(g.GELISIM_1M_UZERI) * 2.5 / 1200,0) AS GELISIMDENPRIM1m             
       
      
                  FROM OFSA_SB.OFSA_TEMP_MEVD_PRIM_2021 g,
                       ( SELECT b.*,a.org_type_desc SUBE_TIPI
                           FROM skzn_prod.skzn_pr_org_type_desc@qadata_link a,skzn_prod.skzn_pr_org_type@qadata_link b
                          WHERE b.org_type=a.org_type )SUBE_TIPI --SKZN QADATA geçişi sebebiyle tablolar linkli hale getirilmiştir. Y.T. 02.09.2024

          WHERE g.as_of_date = PDATE 
            and g.sube=SUBE_TIPI.ORG_CODE
            and g.PB='TL'
            and g.GELISIM_VADESIZ > 0

           CASE WHEN TO_DATE (D.MATURITY_DATE, 'yyyymmdd') < LAST_DAY(PDATE)

    WHERE MATURITY_DATE IS NOT NULL

                          AND RECLASS_CODE_PRV2 IN ('G', 'Z')
                          /*AND SUBSTR(ACCOUNT_NO, 1, 6) IN
                            ('547311',
                              '403836',
                              '411159',
                              '519753')*/                          
                          ) KART                      
                      LEFT JOIN                      
                      (SELECT BRANCH_CODE,
                          ACCOUNT_NO,
                          RECEIVER_NAME,
                          COURIER_COMMENT,
                          TESLIM_TARIHI TESLIM_TARIHI,
                          ONCEKIAYSONU,
                          ONCEKIAYSONU -
                          TO_DATE(TESLIM_TARIHI, 'yyyymmdd') BEKLEME_GUN                      
                      FROM 
                          (SELECT B.ACCOUNT_NO,
                              B.BRANCH_CODE,
                              NVL(A.ARRIVAL_DATE,
                                A.LAST_DELIVERY_DATE) TESLIM_TARIHI,
                              A.COURIER_COMMENT,
                              A.RECEIVER_NAME,
                              A.COURIER_DETAIL_STATUS,
                              A.COURIER_MAIN_STATUS,
                              A.COST_NOTICE_DATE,
                              B.CARD_TRACKING_STATUS,
                              RANK() OVER(PARTITION BY B.ACCOUNT_NO ORDER BY NVL(A.ARRIVAL_DATE, A.LAST_DELIVERY_DATE) DESC) AS SIRA1
                         FROM CARD.CCC_CARD_DELIVERY_ACTIONS@CBRAS_PRODUX A,
                              CARD.CMS_ACCOUNT@CBRAS_PRODUX               B,
                              CARD.CMS_CARD_ISSUE@CBRAS_PRODUX            C,
                              CARD.CMS_ACCOUNT_CARD@CBRAS_PRODUX          D
                          WHERE B.OID(+) = D.ACCOUNT_OID
                            AND C.ACCOUNT_CARD_OID = D.OID(+)
                            AND B.STATUS = 1
                            AND A.CARD_ISSUE_OID(+) = C.OID
                            AND A.COURIER_DETAIL_STATUS IN
                              ('11907')
                          )
                      WHERE SIRA1 = 1) KURYE
                        ON KURYE.ACCOUNT_NO = KART.ACCOUNT_NO
                                           )                  
                  WHERE BEKLEME_GUN >= 30                  
                  )              
              )
          GROUP BY BRANCH_CODE, CUSTOMER_CODE          
          UNION ALL          
          SELECT 'ŞUBEDE_BEKLEYEN_CEK_KARNE' AS TUR,
            BRANCH_CODE,
            CUSTOMER_CODE,
            COUNT(CUSTOMER_CODE) ADET,
            COUNT(CUSTOMER_CODE) * 250 TUTAR
          FROM (SELECT A.CUSTOMER_CODE,
                A.CHECK_COUNT,
                A.ACCOUNT_NO,
                A.STATE,
                A.ISLEM_TARIHI,
                A.BRANCH_CODE,
                A.CHECKBOOK_NO,
                A.KODU,
                A.ADI              
              FROM OLAP.DM_ORGANIZATION O,                
                (SELECT C.CUSTOMER_CODE,
                    NVL((CUS.TITLE),
                      (CUS.NAME || '  ' || CUS.SECOND_NAME || '  ' ||
                      CUS.SURNAME)),
                    B.CHECK_COUNT,
                    B.CURRENCY_CODE,
                    B.ACCOUNT_NO,
                    B.STATE,
                    TO_DATE(D.TRANSACTION_DATE, 'yyyymmdd') ISLEM_TARIHI,
                    LOOKUP_VALUES4.CODE KODU,
                    LOOKUP_VALUES4.NAME ADI,
                    B.BRANCH_CODE,
                    O.NAME,
                    D.USER_NO,
                    D.REFERENCE_ID,
                    DECODE(B.CHECKBOOK_TYPE,
                        '1',
                        'Normal Çek',
                        '2',
                        'Continuous Çek',
                        '3',
                        'Çizgili Blokeli Şube Çek',
                        '4',
                        'Gümüş Üreten Çek',
                        '5',
                        'Altın Üreten Çek',
                        '6',
                        'Gümüş Prestij Çek',
                        '7',
                        'Altın Prestij Çek',
                        '8',
                        'Platinum Prestij Çek'),
                    B.CHECKBOOK_NO,
                    B.MIN_CHECK_NO,
                    B.MAX_CHECK_NO,
                    INFRA_CUST_CUST_PROFIT_CENTER2.PROFIT_CENTER_NAME,
                    R.CUSTOMER_RISK_CODE,
                    INFRA_CUST_CUST_PROFIT_CENTER2.PROFIT_SEGMENT_NAME,
                    T.PARENT_ORG_NAME,
                    (CASE
                      WHEN B.CHECKBOOK_TYPE = '2' THEN
                      DECODE(B.IS_LASER, '1', 'Evet', 'Hayır')
                      ELSE
                      ''
                    END),
                    (CASE
                      WHEN B.CHECKBOOK_TYPE = '3' THEN
                      ''
                      ELSE
                      DECODE(B.IS_QR_CODE, '1', 'Evet', 'Hayır')
                    END)
                  FROM CHECKSBILLS.CHECK_CHECKBOOK_OWNER@CBRAS_PRODUX C,
                    INFRA.CUST_CUST_CUST@CBRAS_PRODUX CUS,
                    CHECKSBILLS.CHECK_CHECKBOOK@CBRAS_PRODUX B,
                    CHECKSBILLS.CHECK_CHECKBOOK_TRANSACTIONS@CBRAS_PRODUX D,
                    (SELECT U.CODE CODE, U.NAME NAME, Y.NAME PRM
                      FROM AURORA.DICT_ML_REFERENCE_DATA_ITEM@CBRAS_PRODUX U,
                        AURORA.DICT_ML_REFERENCE_DATA@CBRAS_PRODUX      Y
                      WHERE U.REFERENCE_DATA_OID = Y.OID) LOOKUP_VALUES4,
                    INFRA.ADMIN_ORG_ORGANIZATION@CBRAS_PRODUX O,
                    INFRA.CUST_CUST_PROFIT_CENTER@CBRAS_PRODUX INFRA_CUST_CUST_PROFIT_CENTER2,
                    CCS.ARL_CMMN_CUSTOMER_RISK_CODE@CBRAS_PRODUX R,
                    (SELECT SUBE.*,
                        SUBE.CODE           AS PARENT_ORG_CODE,
                        SUBE.NAME           AS PARENT_ORG_NAME,
                        SUBE.CODE           AS SUBE_CODE,
                        DEP.DEPENDENCE_NAME
                      FROM INFRA.ADMIN_ORG_ORGANIZATION@CBRAS_PRODUX   SUBE,
                        INFRA.ADMIN_ORG_DEPENDENCE@CBRAS_PRODUX     DEP,
                        INFRA.ADMIN_ORG_ORG_DEPENDENCE@CBRAS_PRODUX ORG_DEP
                      WHERE DEPENDENCE_CODE = '014'
                      AND SUBE.OID = ORG_DEP.ORGANIZATION_OID(+)
                      AND DEP.OID(+) = ORG_DEP.DEPENDENCE_OID                      
                      ) T,                    
                    DEPOSIT.DEPOSIT_ACCOUNT@CBRAS_PRODUX DEPOSIT
                  WHERE B.OID = C.CHECKBOOK_OID
                  AND C.CUSTOMER_CODE = CUS.CUSTOMER_CODE
                  AND B.OID = D.CHECKBOOK_OID(+)
                  AND B.OWNER_OID = C.OID
                  AND LOOKUP_VALUES4.PRM =
                    'PRM_CHECKS_BOOKMNG_BOOK_STATE'
                  AND LOOKUP_VALUES4.CODE = B.STATE
                  AND O.CODE = B.BRANCH_CODE
                  AND O.STATUS = 1
                  AND DEPOSIT.STATUS = 1
                  AND DEPOSIT.OID = C.ACCOUNT_OID
                  AND CUS.PROFIT_CENTER_CODE =
                    INFRA_CUST_CUST_PROFIT_CENTER2.PROFIT_CENTER_CODE(+)
                  AND CUS.PROFIT_SEGMENT_CODE =
                    INFRA_CUST_CUST_PROFIT_CENTER2.PROFIT_SEGMENT_CODE(+)
                  AND INFRA_CUST_CUST_PROFIT_CENTER2.STATUS(+) = 1
                  AND R.CUSTOMER_CODE = C.CUSTOMER_CODE
                  AND R.STATUS = 1
                  AND DEPOSIT.ORG_CODE = T.CODE
                  AND B.STATUS = '1'
                    --TO_DATE(D.TRANSACTION_DATE,'yyyymmdd')  BETWEEN  
                  AND D.STATE = '01'                  
                  ) A              
              WHERE A.KODU = '10'
              AND O.GL_ORG_CODE = A.BRANCH_CODE
              AND O.STATE = 1
              AND A.ISLEM_TARIHI <= PDATE - 30)
          GROUP BY BRANCH_CODE, CUSTOMER_CODE          
          ) TABLO,
        OLAP.FC_CUSTOMER C,
        OLAP.DM_ORGANIZATION ORG
      WHERE TABLO.CUSTOMER_CODE = C.CUSTOMER_CODE
      AND TABLO.BRANCH_CODE = ORG.GL_ORG_CODE
      AND C.AS_OF_DATE = SEGM_TARIH      
      GROUP BY           
          CASE
            WHEN TO_NUMBER(TABLO.BRANCH_CODE) IN
              ('6', '122',  '220',   '372') THEN
            -1
            ELSE
            TO_NUMBER(ORG.PARENT_CODE)
          END,
          TO_NUMBER(TABLO.BRANCH_CODE),
          TABLO.TUR
          );
     
     END IF;
     
     COMMIT;

    /*****************************************************************************************************************************/      
   OFSA_SB.MN_UTILITY.UPDATE_PLOG(PID => PID, SUCCESS => 1, PCRITICAL => 9);

  EXCEPTION
    WHEN OTHERS THEN
      OFSA_SB.MN_UTILITY.UPDATE_PLOG(PID => PID,
                                   SUCCESS => 0,
                                   FAIL => 'POSITION: ' || ERROR_LOCATION || ' ERROR: ' || ABS(SQLCODE) || ':' || SQLERRM || ' ' || DBMS_UTILITY.FORMAT_ERROR_BACKTRACE,
                                   PCRITICAL => PCRITICAL_LVL);
      IF ABS(SQLCODE) = '08103'
         OR ABS(SQLCODE) = '12801'     
                               
      THEN
        ROLLBACK;
--        DBMS_LOCK.SLEEP(180);
         OFSA_SB.MS_EX_OFSA_SB_MANUEL_KALEMLER.MS_EX_OFSA_SB_MANUEL_KALEMLER();
      ELSE
      ROLLBACK;
        RAISE;
        END IF;
  END;

END;
