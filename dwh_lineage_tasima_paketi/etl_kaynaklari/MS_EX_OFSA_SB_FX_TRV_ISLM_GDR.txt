PROCEDURE MS_EX_OFSA_SB_FX_TRV_ISLM_GDR (SECIM NUMBER)IS

    ERROR_LOCATION     NUMBER;
    PID                NUMBER;
    PDATE              DATE;
    AYBASI             DATE;
    PASOFDATE          VARCHAR2(8);
    PCRITICAL_LVL      NUMBER := 1; --1:BLOCKER,2:CRITICAL,3:TRIVIAL,9:NO PROBLEM
    SEGM_TARIH         DATE;
    READY              NUMBER;
BEGIN

    ERROR_LOCATION := 1;

   SELECT PRM_VALUE, TO_DATE(PRM_VALUE, PRM_FORMAT), TO_DATE(MONTH_BEGIN, PRM_FORMAT),TO_DATE(LAST_SEGM_DATE, PRM_FORMAT)
      INTO PASOFDATE, PDATE,AYBASI,SEGM_TARIH
      FROM OFSA_SB.SY_SYSTEM_PARAMETERS T
      WHERE PARAMETER = 'OFSA_SB_DATE';

      IF PDATE<=SEGM_TARIH THEN --SEGMENTASYON ATAMASI
      SEGM_TARIH:=SEGM_TARIH;
      ELSE
      SEGM_TARIH:=PDATE;
      END IF;

    PID := OFSA_SB.MN_UTILITY.INSERT_PLOG(PMODULE      => 'MS_EX_OFSA_SB_FX_TRV_ISLM_GDR',
                                          PSUBMODULE   => 'EX_OFSA_SB_FX_TRV_ISLM_GDR',
                                          PROJECT_NAME => 'OFSA_SB',
                                          PASOFDATE    => PDATE,
                                          PCRITICAL    => PCRITICAL_LVL);
   /*****************************************************************************************************************************/
   Loop
        SELECT count(*)
        into READY
        FROM DM.OFSA_DAILY_INVESTMENT@QADATA_LINK A
        where as_of_date=PDATE
        AND PRODUCT_TYPE = 'BFX';
        exit when READY>0;
        sleep(300000);--5dk
        end loop;
   /*****************************************************************************************************************************/

    EXECUTE IMMEDIATE 'ALTER SESSION ENABLE PARALLEL DML';
    
    EXECUTE IMMEDIATE 'ALTER session set nls_date_format= '||'yyyymmdd'||''; 

    IF SECIM=0 THEN --Eger 0 seçilirse günlük,0 dışında bir veri seçilirse aylık yükler
     AYBASI:=PDATE;
    END IF;

     (
    SELECT PDATE AS AS_OF_DATE,
       NVL(MUS_NO,999999999) AS MUS_NO,
       ISKOLU AS ISKOLU,
       SEGMENT AS SEGMENT,
       CASE WHEN TO_NUMBER(SUBE) IN ('6','122', '220','372') THEN -1 ELSE TO_NUMBER(O.PARENT_CODE) END BOLGE,
       TO_NUMBER(SUBE) AS SUBE,
       NVL(MT,88888) AS MT,
       '88888888' AS URUN_KODU,
       URUN,
       CCY AS PB,
       'FX_HAZINE_GELIR_GIDER' AS KAYNAK,
       SUM(BAKIYE) BAKIYE
    FROM OLAP.DM_ORGANIZATION O,
       (SELECT PDATE AS TARIH,
               CASE WHEN A.CUS_NO IS NULL THEN TO_NUMBER(A.ORG_CODE) ELSE TO_NUMBER(C.MAIN_BRANCH_CODE) END AS SUBE,
               A.CUS_NO AS MUS_NO,
               A.CUS_NAME AS MUS_ADI,
               'BFX' AS URUN,
               OFSA_SB.MN_UTILITY.MAP_OFSA_ISKOLU (BUSINESS_SEGMENT_CODE   => SEG.BUSINESS_SEGMENT_CODE) AS ISKOLU,
               SEG.BUSINESS_SEGMENT_CODE AS SEGMENT,
               A.ISO_CURRENCY_CD CCY,
               NVL(C.REPRESENTATIVE_FILE_NO, 999999) AS MT,
               SUM(A.FX_GAIN_LOSS) AS BAKIYE
          FROM DM.OFSA_DAILY_INVESTMENT@QADATA_LINK A, OLAP.FC_CUSTOMER C, OLAP.FC_CUSTOMER SEG
         WHERE A.AS_OF_DATE BETWEEN  AYBASI AND PDATE
           AND C.AS_OF_DATE (+) = PDATE
           AND A.CUS_NO  = C.CUSTOMER_CODE (+)
           AND SEG.AS_OF_DATE (+) = SEGM_TARIH
           AND A.CUS_NO = SEG.CUSTOMER_CODE (+)
           AND PRODUCT_TYPE = 'BFX'
          -- AND NVL(A.CUS_NO,0)!=0
           AND A.FX_GAIN_LOSS <> 0
           AND A.ID_NUMBER NOT LIKE 'BFXDCD%'
           AND A.ID_NUMBER NOT LIKE 'BFXFW%'        
         GROUP BY CASE WHEN A.CUS_NO IS NULL THEN TO_NUMBER(A.ORG_CODE) ELSE TO_NUMBER(C.MAIN_BRANCH_CODE) END, A.CUS_NO, A.CUS_NAME,
            OFSA_SB.MN_UTILITY.MAP_OFSA_ISKOLU (BUSINESS_SEGMENT_CODE   => SEG.BUSINESS_SEGMENT_CODE),
                  SEG.BUSINESS_SEGMENT_CODE,
                  A.ISO_CURRENCY_CD,
                  NVL(C.REPRESENTATIVE_FILE_NO, 999999)
        UNION ALL
        SELECT PDATE AS TARIH,

           AND C.AS_OF_DATE = PDATE
           AND SEG.CUSTOMER_CODE = A.CUST_NO

                  SEG.BUSINESS_SEGMENT_CODE,

                       TO_NUMBER(A.CUST_NO) AS MUS_NO,
                       A.CUST_NAME AS MUS_ADI,
                       'FORWARD' AS URUN,
                       OFSA_SB.MN_UTILITY.MAP_OFSA_ISKOLU (BUSINESS_SEGMENT_CODE   => SEG.BUSINESS_SEGMENT_CODE) AS ISKOLU,
                       SEG.BUSINESS_SEGMENT_CODE SEGMENT,
                       NVL(C.REPRESENTATIVE_FILE_NO, 999999) AS MT,
                       A.CURRENCY_CODE CCY,

                    --   ROUND(SUM(A.PROFIT_LOSS /
                    --             POWER((1 + LIB.ORAN / 36500),
                    --                   (TO_NUMBER(TO_DATE(A.RETURN_DATE,'YYYYMMDD') - PDATE + 1)))),

                 WHERE TO_DATE(A.VALOR_DATE,'YYYYMMDD') BETWEEN AYBASI AND PDATE

                --   AND TO_DATE(LIB.RATE_DATE,'YYYYMMDD') = PDATE

                   SEG.BUSINESS_SEGMENT_CODE,
                   NVL(C.REPRESENTATIVE_FILE_NO, 999999),
                   M.BUY_CURRENCY_CODE                   
         UNION ALL         
           SELECT PDATE AS TARIH,
                      TO_NUMBER(C.MAIN_BRANCH_CODE) AS SUBE,

              AND A.TUREV_PURPOSE='1'
              AND (ABS(A.TREASURE_DIRTY_UNIT_PRICE-A.DIRTY_UNIT_PRICE)*A.NOMINAL*A.EXCHANGE_RATE/100)<>0
              AND A.INV_ACC_CUST_CODE!=1
              AND C.CUSTOMER_CODE = A.INV_ACC_CUST_CODE
              AND C.AS_OF_DATE = PDATE
              AND SEG.CUSTOMER_CODE = A.INV_ACC_CUST_CODE
              AND SEG.AS_OF_DATE = SEGM_TARIH   
              AND TO_NUMBER(C.MAIN_BRANCH_CODE)= DO.ORG_CODE
         GROUP BY TO_NUMBER(C.MAIN_BRANCH_CODE),
                  PDATE,

            )
          WHERE TO_NUMBER(O.GL_ORG_CODE) = TO_NUMBER(SUBE)
          GROUP BY MUS_NO, ISKOLU, SEGMENT, TO_NUMBER(SUBE), MT,
          CASE WHEN TO_NUMBER(SUBE) IN ('6','122', '220','372') THEN -1 ELSE TO_NUMBER(O.PARENT_CODE) END,URUN,CCY
